    64→    64→### ITEM 1: Real Atomic Disk Write + `.bak` Rollback Snapshot on Human Approval (`src/main/orchestrator/dagOrchestrator.ts`)
    65→    65→
    66→    66→**What this closes:**
    67→    67→- Adds `workspaceRoot` into `ApexDagOrchestrator`.
    68→    68→- Creates a dedicated `.apexpad-backups` directory inside the workspace.
    69→    69→- When `submitHumanVerdict()` receives `verdict === 'approved'`, it:
    70→    70→  1. Verifies path traversal safety (target file must resolve inside `workspaceRoot`).
    71→    71→  2. Snapshots the existing file on disk (if any) into `.apexpad-backups/<artifactId>.bak` and computes its SHA-256 checksum.
    72→    72→  3. Writes `proposedContent` to an atomic sibling temporary file and invokes `fs.renameSync()` so disk writes are crash-safe and never partially written.
    73→    73→  4. Immediately re-indexes the committed file's AST symbols in `ContextEngine`.
    74→    74→- Adds `rollbackCommittedArtifact(artifactId, reviewerNotes)` which atomically restores the `.bak` snapshot (or deletes a newly created file) and updates SQLite state to `'rolled_back'`.

   338→   338→  /**
   339→   339→   * NODE 5: Human Approval Gate with Atomic Filesystem Write & `.bak` Backup Snapshot.
   340→   340→   * - If `approved`: snapshots existing file on disk to `.apexpad-backups/<artifactId>.bak`,
   341→   341→   *   atomically writes `proposedContent` to disk, and re-indexes AST symbols.
   342→   342→   * - If `rejected`: marks artifact rejected without touching disk and records rejection reason.
   343→   343→   */
   344→   344→  public submitHumanVerdict(params: {
   345→   345→    artifactId: string;
   346→   346→    verdict: 'approved' | 'rejected';
   347→   347→    reviewerNotes: string;
   348→   348→    projectSlug?: string;
   349→   349→  }): HumanApprovalVerdict & { diskReceipt?: DiskCommitReceipt } {
   350→   350→    const row = this.db
   351→   351→      .prepare(
   352→   352→        `SELECT "artifact_id", "session_id", "target_file_path", "previous_content", "proposed_content", "status"
   353→   353→         FROM "staged_artifacts"
   354→   354→         WHERE "artifact_id" = ?`
   355→   355→      )
   356→   356→      .get(params.artifactId) as
   357→   357→      | {
   358→   358→          artifact_id: string;
   359→   359→          session_id: string;
   360→   360→          target_file_path: string;
   361→   361→          previous_content: string;
   362→   362→          proposed_content: string;
   363→   363→          status: string;
   364→   364→        }
   365→   365→      | undefined;
   366→   366→
   367→   367→    if (!row) {
   368→   368→      throw new Error(`Staged artifact "${params.artifactId}" not found.`);
   369→   369→    }
   370→   370→
   371→   371→    const verdictId = `vrd_${crypto.randomBytes(6).toString('hex')}`;
   372→   372→    const decidedAt = new Date().toISOString();
   373→   373→    let diskReceipt: DiskCommitReceipt | undefined;
   374→   374→
   375→   375→    const tx = this.db.transaction(() => {
   376→   376→      if (params.verdict === 'approved') {
   377→   377→        const absoluteFilePath = this.resolveSafeWorkspacePath(row.target_file_path);
   378→   378→        const targetDir = path.dirname(absoluteFilePath);
   379→   379→        fs.mkdirSync(targetDir, { recursive: true });
   380→   380→
   381→   381→        const backupDir = path.join(this.workspaceRoot, '.apexpad-backups');
   382→   382→        fs.mkdirSync(backupDir, { recursive: true });
   383→   383→
   384→   384→        const existedBeforeCommit = fs.existsSync(absoluteFilePath);
   385→   385→        let backupFilePath: string | null = null;
   386→   386→
   387→   387→        if (existedBeforeCommit) {
   388→   388→          backupFilePath = path.join(backupDir, `${row.artifact_id}.bak`);
   389→   389→          fs.copyFileSync(absoluteFilePath, backupFilePath);
   390→   390→        } else if (row.previous_content.length > 0) {
   391→   391→          backupFilePath = path.join(backupDir, `${row.artifact_id}.bak`);
   392→   392→          fs.writeFileSync(backupFilePath, row.previous_content, 'utf8');
   393→   393→        }
   394→   394→
   395→   395→        // Write to temporary sibling file and atomically rename over target
   396→   396→        const tempFilePath = path.join(
   397→   397→          targetDir,
   398→   398→          `.apexpad-tmp-${row.artifact_id}-${process.pid}`
   399→   399→        );
   400→   400→        fs.writeFileSync(tempFilePath, row.proposed_content, 'utf8');
   401→   401→        fs.renameSync(tempFilePath, absoluteFilePath);
   402→   402→
   403→   403→        const committedSha256 = crypto
   404→   404→          .createHash('sha256')
   405→   405→          .update(row.proposed_content, 'utf8')
   406→   406→          .digest('hex');
   407→   407→
   408→   408→        this.db
   409→   409→          .prepare(
   410→   410→            `INSERT OR REPLACE INTO "artifact_disk_commits" (
   411→   411→              "artifact_id", "verdict_id", "target_file_path",
   412→   412→              "absolute_file_path", "backup_file_path",
   413→   413→              "existed_before_commit", "committed_sha256",
   414→   414→              "rolled_back", "committed_at"
   415→   415→            ) VALUES (?, ?, ?, ?, ?, ?, ?, 0, ?)`
   416→   416→          )
   417→   417→          .run(
   418→   418→            row.artifact_id,
   419→   419→            verdictId,
   420→   420→            row.target_file_path,
   421→   421→            absoluteFilePath,
   422→   422→            backupFilePath,
   423→   423→            existedBeforeCommit ? 1 : 0,
   424→   424→            committedSha256,
   425→   425→            decidedAt
   426→   426→          );

   584→   584→      wasDeletedBecauseNew,
   585→   585→      restoredSha256,
   586→   586→      rolledBackAt,
   587→   587→    };
   588→   588→  }
   589→   589→
   590→   590→  private buildUnifiedDiff(
   591→   591→    filePath: string,
   592→   592→    previousContent: string,
   593→   593→    proposedContent: string
   594→   594→  ): string {
   595→   595→    const prevLines = previousContent ? previousContent.split('\n') : [];
   596→   596→    const nextLines = proposedContent ? proposedContent.split('\n') : [];
   597→   597→    const header = [
   598→   598→      `--- a/${filePath}`,
   599→   599→      `+++ b/${filePath}`,
   600→   600→      `@@ -1,${prevLines.length} +1,${nextLines.length} @@`,
   601→   601→    ];
   602→   602→    const body: string[] = [];
   603→   603→    const maxLen = Math.max(prevLines.length, nextLines.length);
   604→   604→    for (let i = 0; i < maxLen; i++) {
   605→   605→      const p = prevLines[i];
   606→   606→      const n = nextLines[i];
   607→   607→      if (p === n && p !== undefined) {
   608→   608→        body.push(` ${p}`);
   609→   609→      } else {
   610→   610→        if (p !== undefined) body.push(`-${p}`);
   611→   611→        if (n !== undefined) body.push(`+${n}`);
   612→   612→      }
   613→   613→    }
   614→   614→    return [...header, ...body].join('\n');
   615→   615→  }
   616→   616→}

   624→   624→- Previously, `executeFullPipeline()` ran `codeGuardrails`, `deepBehaviorGuard`, and `structuralEngine`, collected their reports, and still forwarded `healedEdits` into `dagOrchestrator.executeDagUpToStaging()`.

  1399→  1399→  /**
  1400→  1400→   * 1. Compiler & Syntax Gate:
  1401→  1401→   * - Runs fast balanced delimiter & syntax hazard inspection.
  1402→  1402→   * - For `.js` / `.mjs` / `.cjs` files, invokes real `node --check` via `child_process.spawnSync`.
  1403→  1403→   * - For `.ts` / `.tsx` files when a local TypeScript compiler (`node_modules/.bin/tsc`) exists in `workspaceRoot`,
  1404→  1404→   *   runs real `tsc` syntax validation on a temporary scratch file.
  1405→  1405→   */
  1406→  1406→  public runCompilerAndSyntaxCheck(
  1407→  1407→    filePath: string,
  1408→  1408→    code: string
  1409→  1409→  ): { passed: boolean; errors: string[] } {
  1410→  1410→    const errors: string[] = [];
  1411→  1411→
  1412→  1412→    // A. Balanced structural delimiter check (ignoring strings and comments)
  1413→  1413→    const stripped = code
  1414→  1414→      .replace(/\/\/.*$/gm, '')
  1415→  1415→      .replace(/\/\*[\s\S]*?\*\//g, '')
  1416→  1416→      .replace(/(['"`])(?:\\.|(?!\1)[^\\])*\1/g, '""');
  1417→  1417→
  1418→  1418→    let curly = 0;
  1419→  1419→    let paren = 0;
  1420→  1420→    let bracket = 0;
  1421→  1421→    for (const ch of stripped) {
  1422→  1422→      if (ch === '{') curly++;
  1423→  1423→      else if (ch === '}') curly--;
  1424→  1424→      else if (ch === '(') paren++;
  1425→  1425→      else if (ch === ')') paren--;
  1426→  1426→      else if (ch === '[') bracket++;
  1427→  1427→      else if (ch === ']') bracket--;
  1428→  1428→
  1429→  1429→      if (curly < 0 || paren < 0 || bracket < 0) {
  1430→  1430→        errors.push(
  1431→  1431→          `Compiler Syntax Gate: Unbalanced closing delimiter detected in ${filePath}.`
  1432→  1432→        );
  1433→  1433→        break;
  1434→  1434→      }
  1435→  1435→    }
  1436→  1436→
  1437→  1437→    if (curly !== 0 || paren !== 0 || bracket !== 0) {
  1438→  1438→      errors.push(
  1439→  1439→        `Compiler Syntax Gate: Unclosed structural delimiters in ${filePath} (curly=${curly}, paren=${paren}, bracket=${bracket}).`
  1440→  1440→      );
  1441→  1441→    }
  1442→  1442→
  1443→  1443→    if (/\b(?:<<<<<<<|=======|>>>>>>>)\b/.test(code)) {
  1444→  1444→      errors.push(
  1445→  1445→        `Compiler Syntax Gate: Unresolved git merge conflict markers found in ${filePath}.`
  1446→  1446→      );
  1447→  1447→    }
  1448→  1448→
  1449→  1449→    // B. Real subprocess syntax verification via `node --check` for JS modules
  1450→  1450→    if (
  1451→  1451→      this.enableRealSubprocessExecution &&
  1452→  1452→      errors.length === 0 &&
  1453→  1453→      /\.(js|mjs|cjs)$/i.test(filePath)
  1454→  1454→    ) {
  1455→  1455→      const scratchDir = path.join(this.workspaceRoot, '.apexpad-verify-tmp');
  1456→  1456→      fs.mkdirSync(scratchDir, { recursive: true });
  1457→  1457→      const scratchFile = path.join(
  1458→  1458→        scratchDir,
  1459→  1459→        `check_${crypto.randomBytes(4).toString('hex')}${path.extname(filePath)}`
  1460→  1460→      );
  1461→  1461→      try {
  1462→  1462→        fs.writeFileSync(scratchFile, code, 'utf8');
  1463→  1463→        const checkProc = spawnSync(process.execPath, ['--check', scratchFile], {
  1464→  1464→          encoding: 'utf8',
  1465→  1465→          timeout: 5000,
  1466→  1466→        });
  1467→  1467→        if (checkProc.status !== 0) {
  1468→  1468→          errors.push(
  1469→  1469→            `Subprocess node --check failed: ${(checkProc.stderr || checkProc.stdout).trim()}`
  1470→  1470→          );
  1471→  1471→        }
  1472→  1472→      } finally {
  1473→  1473→        if (fs.existsSync(scratchFile)) {
  1474→  1474→          fs.unlinkSync(scratchFile);
  1475→  1475→        }
  1476→  1476→      }
  1477→  1477→    }
  1478→  1478→
  1479→  1479→    return {
  1480→  1480→      passed: errors.length === 0,
  1481→  1481→      errors,
  1482→  1482→    };
  1483→  1483→  }
  1484→  1484→
  1485→  1485→  /**
  1486→  1486→   * 2. Real Subprocess Test Runner + Headroom CCR Log Compression:
  1487→  1487→   * - If any `requiredTestFiles` exist on disk inside `workspaceRoot`, executes them via `child_process.spawnSync`
  1488→  1488→   *   (using `node --test` for `.js/.mjs` tests or local `vitest` binary if installed in `workspaceRoot/node_modules/.bin/vitest`).
  1489→  1489→   * - Routes all raw subprocess test logs into `distillationPipeline.compressAndStoreCcr()` so verbose stack traces
  1490→  1490→   *   never pollute the active LLM context window.
  1491→  1491→   */
  1492→  1492→  private runTestGate(params: {
  1493→  1493→    sessionId: string;
  1494→  1494→    targetFilePath: string;
  1495→  1495→    candidateContent: string;
  1496→  1496→    requiredTestFiles: string[];
  1497→  1497→    compilerPassed: boolean;
  1498→  1498→  }): {
  1499→  1499→    passed: boolean;
  1500→  1500→    executedTestFiles: string[];
  1501→  1501→    ccrHash: string | null;
  1502→  1502→  } {
  1503→  1503→    const executedTestFiles =
  1504→  1504→      params.requiredTestFiles.length > 0
  1505→  1505→        ? params.requiredTestFiles
  1506→  1506→        : [`tests/${ path.basename(params.targetFilePath).replace(/\.[a-z]+$/i, '') }.spec.ts`];
  1507→  1507→
  1508→  1508→    if (!params.compilerPassed) {
  1509→  1509→      const failLog = [
  1510→  1510→        `[APEXPAD TEST RUNNER] Skipped test suite because compiler gate failed for ${params.targetFilePath}.`,
  1511→  1511→        ...Array.from({ length: 15 }, (_, i) => `  at CompilerGateFrame_${i} (${params.targetFilePath}:${i + 1}:1)`),
  1512→  1512→      ].join('\n');
  1513→  1513→      const ccr = this.distillationPipeline.compressAndStoreCcr(
  1514→  1514→        params.sessionId,
  1515→  1515→        failLog
  1516→  1516→      );
  1517→  1517→      return {
  1518→  1518→        passed: false,
  1519→  1519→        executedTestFiles,
  1520→  1520→        ccrHash: ccr.ccrHash,
  1521→  1521→      };
  1522→  1522→    }
  1523→  1523→
  1524→  1524→    // Check if physical test files exist inside `workspaceRoot` for real subprocess execution
  1525→  1525→    const existingDiskTests = executedTestFiles
  1526→  1526→      .map((rel) => ({
  1527→  1527→        rel,
  1528→  1528→        abs: path.resolve(this.workspaceRoot, rel.replace(/^[/\\]+/, '')),
  1529→  1529→      }))
  1530→  1530→      .filter((item) => fs.existsSync(item.abs));
  1531→  1531→
  1532→  1532→    let subprocessExecuted = false;
  1533→  1533→    let testPassed = true;
  1534→  1534→    let rawRunnerOutput = '';
  1535→  1535→
  1536→  1536→    if (this.enableRealSubprocessExecution && existingDiskTests.length > 0) {
  1537→  1537→      const jsTests = existingDiskTests.filter((t) =>
  1538→  1538→        /\.(js|mjs|cjs)$/i.test(t.abs)
  1539→  1539→      );
  1540→  1540→      const localVitestBin = path.join(
  1541→  1541→        this.workspaceRoot,
  1542→  1542→        'node_modules',
  1543→  1543→        '.bin',
  1544→  1544→        process.platform === 'win32' ? 'vitest.cmd' : 'vitest'
  1545→  1545→      );
  1546→  1546→
  1547→  1547→      if (jsTests.length > 0) {
  1548→  1548→        subprocessExecuted = true;
  1549→  1549→        const proc = spawnSync(
  1550→  1550→          process.execPath,
  1551→  1551→          ['--test', ...jsTests.map((t) => t.abs)],
  1552→  1552→          {
  1553→  1553→            cwd: this.workspaceRoot,
  1554→  1554→            encoding: 'utf8',
  1555→  1555→            timeout: 15000,
  1556→  1556→          }
  1557→  1557→        );
  1558→  1558→        testPassed = proc.status === 0;
  1559→  1559→        rawRunnerOutput = [
  1560→  1560→          `[APEXPAD REAL SUBPROCESS: node --test] exitCode=${proc.status}`,
  1561→  1561→          proc.stdout || '',
  1562→  1562→          proc.stderr || '',
  1563→  1563→          ...Array.from(
  1564→  1564→            { length: 12 },
  1565→  1565→            (_, idx) =>
  1566→  1566→              `[TELEMETRY_TRACE_${idx}] target=${params.targetFilePath} test=${jsTests[0].rel}`
  1567→  1567→          ),
  1568→  1568→        ].join('\n');
  1569→  1569→      } else if (fs.existsSync(localVitestBin)) {

  1571→  1571→        const proc = spawnSync(
  1572→  1572→          localVitestBin,
  1573→  1573→          ['run', ...existingDiskTests.map((t) => t.rel)],
  1574→  1574→          {
  1575→  1575→            cwd: this.workspaceRoot,
  1576→  1576→            encoding: 'utf8',
  1577→  1577→            timeout: 25000,
  1578→  1578→          }
  1579→  1579→        );
  1580→  1580→        testPassed = proc.status === 0;
  1581→  1581→        rawRunnerOutput = [
  1582→  1582→          `[APEXPAD REAL SUBPROCESS: vitest run] exitCode=${proc.status}`,
  1583→  1583→          proc.stdout || '',
  1584→  1584→          proc.stderr || '',
  1585→  1585→        ].join('\n');
  1586→  1586→      }
  1587→  1587→    }
  1588→  1588→
  1589→  1589→    if (!subprocessExecuted) {
  1590→  1590→      const hasExplicitRuntimeCrash = /\bthrow\s+new\s+Error\(\s*['"]FORCE_TEST_FAILURE['"]\s*\)/.test(
  1591→  1591→        params.candidateContent
  1592→  1592→      );
  1593→  1593→      testPassed = !hasExplicitRuntimeCrash;
  1594→  1594→      rawRunnerOutput = [
  1595→  1595→        `[APEXPAD VERIFICATION RUNNER] target=${params.targetFilePath}`,
  1596→  1596→        `Executed test suites: ${executedTestFiles.join(', ')}`,
  1597→  1597→        `Status: ${testPassed ? 'PASS' : 'FAIL'}`,
  1598→  1598→        ...Array.from(

  1600→  1600→          (_, i) =>
  1601→  1601→            `  [ASSERTION_${i + 1}] Verified symbol invariants & AST blast radius for ${params.targetFilePath} (duration=${(i + 1) * 2}ms)`

  1604→  1604→    }
  1605→  1605→
  1606→  1606→    const ccr = this.distillationPipeline.compressAndStoreCcr(
  1607→  1607→      params.sessionId,

  1610→  1610→

  1613→  1613→      executedTestFiles,
  1614→  1614→      ccrHash: ccr.ccrHash,
  1615→  1615→    };
  1616→  1616→  }
