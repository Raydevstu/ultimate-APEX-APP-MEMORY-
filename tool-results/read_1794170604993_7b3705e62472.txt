    62→## 2. Step-by-Step Implementation of the 4 Production Wiring Upgrades
    63→
    64→### ITEM 1: Real Atomic Disk Write + `.bak` Rollback Snapshot on Human Approval (`src/main/orchestrator/dagOrchestrator.ts`)
    65→
    66→**What this closes:**
    67→- Adds `workspaceRoot` into `ApexDagOrchestrator`.
    68→- Creates a dedicated `.apexpad-backups` directory inside the workspace.
    69→- When `submitHumanVerdict()` receives `verdict === 'approved'`, it:
    70→  1. Verifies path traversal safety (target file must resolve inside `workspaceRoot`).
    71→  2. Snapshots the existing file on disk (if any) into `.apexpad-backups/<artifactId>.bak` and computes its SHA-256 checksum.
    72→  3. Writes `proposedContent` to an atomic sibling temporary file and invokes `fs.renameSync()` so disk writes are crash-safe and never partially written.
    73→  4. Immediately re-indexes the committed file's AST symbols in `ContextEngine`.
    74→- Adds `rollbackCommittedArtifact(artifactId, reviewerNotes)` which atomically restores the `.bak` snapshot (or deletes a newly created file) and updates SQLite state to `'rolled_back'`.
    75→
    76→```typescript
    77→/**
    78→ * src/main/orchestrator/dagOrchestrator.ts
    79→ * LangGraph-Style Deterministic State Machine + CrewAI Planner/Research/Architect Agents
    80→ * UPGRADED (Production Gap #1 Closed):
    81→ * - Atomic filesystem write (`fs.writeFileSync` temp + `fs.renameSync`) on Human Approval
    82→ * - Pre-commit `.apexpad-backups/<artifactId>.bak` snapshot creation
    83→ * - Full deterministic `rollbackCommittedArtifact()` restoring exact disk state
    84→ */

   187→  private resolveSafeWorkspacePath(relativeFilePath: string): string {
   188→    const normalizedRel = relativeFilePath.replace(/^[/\\]+/, '');
   189→    const resolved = path.resolve(this.workspaceRoot, normalizedRel);
   190→    const rootWithSep = this.workspaceRoot.endsWith(path.sep)
   191→      ? this.workspaceRoot
   192→      : `${this.workspaceRoot}${path.sep}`;
   193→
   194→    if (resolved !== this.workspaceRoot && !resolved.startsWith(rootWithSep)) {
   195→      throw new Error(
   196→        `Security Violation: Target path "${relativeFilePath}" escapes workspace root "${this.workspaceRoot}".`
   197→      );
   198→    }
   199→    return resolved;
   200→  }

   202→  /**
   203→   * Executes the full LangGraph-style DAG from Context Pack assembly down to Artifact Staging.
   204→   */
   205→  public executeDagUpToStaging(params: {
   206→    sessionId: string;
   207→    projectSlug: string;
   208→    userGoal: string;
   209→    tokenBudgetCeiling: number;
   210→    candidateEdits: Array<{
   211→      targetFilePath: string;
   212→      previousContent: string;
   213→      proposedContent: string;
   214→    }>;
   215→  }): DagExecutionState {
   216→    const runId = `dag_${crypto.randomBytes(6).toString('hex')}`;
   217→
   218→    // NODE 1: Assemble Context Pack (Project DNA + Mem0 Index + AST Focal Symbols)
   219→    const dnaRules = this.memoryRouter.getProjectDnaRules(params.projectSlug);
   220→    const memoryHits = this.memoryRouter.searchIndex({
   221→      projectSlug: params.projectSlug,
   222→      query: params.userGoal,
   223→      limit: 8,
   224→    });
   225→    const focalSymbols = this.contextEngine.queryFocalSymbols({
   226→      projectSlug: params.projectSlug,
   227→      searchQuery: params.userGoal,
   228→      limit: 12,
   229→    });
   230→
   231→    const contextPack = this.contextEngine.assembleContextPack({
   232→      projectSlug: params.projectSlug,
   233→      tokenBudgetCeiling: params.tokenBudgetCeiling,
   234→      dnaRules,
   235→      memoryIndexEntries: memoryHits,
   236→      focalSymbols,
   237→      activeTaskPrompt: params.userGoal,
   238→    });
   239→
   240→    // NODE 2: Planner (CrewAI) — Decompose goal into explicit Research & Architect subtasks
   241→    const targetFiles = params.candidateEdits.map((e) => e.targetFilePath);
   242→    const subTasks: DagSubTask[] = [
   243→      {
   244→        taskId: `${runId}_plan`,
   245→        assignedAgent: 'planner_crewai',
   246→        title: 'Decompose goal & bind Project DNA invariants',
   247→        targetFiles,
   248→        status: 'completed',
   249→        outputSummary: `Bound ${dnaRules.length} DNA rules and ${memoryHits.length} Mem0 index pointers within ${contextPack.totalAllocatedTokens}/${contextPack.tokenBudgetCeiling} tokens.`,
   250→      },
   251→      {
   252→        taskId: `${runId}_research`,
   253→        assignedAgent: 'research_agent',
   254→        title: 'Compute AST Blast Radius & Call-Graph Dependents',
   255→        targetFiles,
   256→        status: 'completed',
   257→        outputSummary: `Indexed ${focalSymbols.length} high-PageRank symbols across ${targetFiles.length} target files.`,
   258→      },
   259→      {
   260→        taskId: `${runId}_architect`,
   261→        assignedAgent: 'architect_agent',
   262→        title: 'Validate Architectural Layering & Stage Unified Diffs',
   263→        targetFiles,
   264→        status: 'completed',
   265→        outputSummary: `Prepared ${params.candidateEdits.length} candidate artifact(s) for Human Approval gate.`,
   266→      },
   267→    ];
   268→
   269→    // NODE 3 & 4: Research Agent + Architect Agent -> Artifact Staging
   270→    const stagedArtifacts: StagedArtifact[] = [];
   271→    const now = new Date().toISOString();
   272→
   273→    for (const edit of params.candidateEdits) {
   274→      const blastRadius = this.contextEngine.computeBlastRadius(
   275→        params.projectSlug,
   276→        edit.targetFilePath
   277→      );
   278→      const archViolations = this.contextEngine.validateArchitecturalRules(
   279→        edit.targetFilePath,
   280→        edit.proposedContent,
   281→        dnaRules
   282→      );
   283→      blastRadius.architecturalViolations.push(...archViolations);
   284→
   285→      const unifiedDiff = this.buildUnifiedDiff(
   286→        edit.targetFilePath,
   287→        edit.previousContent,
   288→        edit.proposedContent
   289→      );
   290→
   291→      const artifactId = `stg_${crypto.randomBytes(6).toString('hex')}`;
   292→      const artifact: StagedArtifact = {
   293→        artifactId,
   294→        sessionId: params.sessionId,
   295→        targetFilePath: edit.targetFilePath,
   296→        previousContent: edit.previousContent,
   297→        proposedContent: edit.proposedContent,
   298→        unifiedDiff,
   299→        blastRadius,
   300→        status: 'staged',
   301→        createdAt: now,
   302→      };
   303→
   304→      this.db
   305→        .prepare(
   306→          `INSERT INTO "staged_artifacts" (
   307→            "artifact_id", "session_id", "target_file_path",
   308→            "previous_content", "proposed_content", "unified_diff",
   309→            "blast_radius_json", "status", "created_at"
   310→          ) VALUES (?, ?, ?, ?, ?, ?, ?, 'staged', ?)`
   311→        )
   312→        .run(
   313→          artifact.artifactId,
   314→          artifact.sessionId,
   315→          artifact.targetFilePath,
   316→          artifact.previousContent,
   317→          artifact.proposedContent,
   318→          artifact.unifiedDiff,
   319→          JSON.stringify(artifact.blastRadius),
   320→          artifact.createdAt
   321→        );
   322→
   323→      stagedArtifacts.push(artifact);
   324→    }
   325→
   326→    return {
   327→      runId,
   328→      sessionId: params.sessionId,
   329→      projectSlug: params.projectSlug,
   330→      userGoal: params.userGoal,
   331→      currentNode: 'human_approval',
   332→      contextPack,
   333→      subTasks,
   334→      stagedArtifacts,
   335→    };
   336→  }
   337→
   338→  /**
   339→   * NODE 5: Human Approval Gate with Atomic Filesystem Write & `.bak` Backup Snapshot.
   340→   * - If `approved`: snapshots existing file on disk to `.apexpad-backups/<artifactId>.bak`,
   341→   *   atomically writes `proposedContent` to disk, and re-indexes AST symbols.
   342→   * - If `rejected`: marks artifact rejected without touching disk and records rejection reason.
   343→   */
   344→  public submitHumanVerdict(params: {
   345→    artifactId: string;
   346→    verdict: 'approved' | 'rejected';
   347→    reviewerNotes: string;
   348→    projectSlug?: string;
   349→  }): HumanApprovalVerdict & { diskReceipt?: DiskCommitReceipt } {
   350→    const row = this.db
   351→      .prepare(
   352→        `SELECT "artifact_id", "session_id", "target_file_path", "previous_content", "proposed_content", "status"
   353→         FROM "staged_artifacts"
   354→         WHERE "artifact_id" = ?`
   355→      )
   356→      .get(params.artifactId) as
   357→      | {
   358→          artifact_id: string;
   359→          session_id: string;
   360→          target_file_path: string;
   361→          previous_content: string;
   362→          proposed_content: string;
   363→          status: string;
   364→        }
   365→      | undefined;
   366→
   367→    if (!row) {
   368→      throw new Error(`Staged artifact "${params.artifactId}" not found.`);
   369→    }
   370→
   371→    const verdictId = `vrd_${crypto.randomBytes(6).toString('hex')}`;
   372→    const decidedAt = new Date().toISOString();
   373→    let diskReceipt: DiskCommitReceipt | undefined;
   374→
   375→    const tx = this.db.transaction(() => {
   376→      if (params.verdict === 'approved') {
   377→        const absoluteFilePath = this.resolveSafeWorkspacePath(row.target_file_path);
   378→        const targetDir = path.dirname(absoluteFilePath);
   379→        fs.mkdirSync(targetDir, { recursive: true });
   380→
   381→        const backupDir = path.join(this.workspaceRoot, '.apexpad-backups');
   382→        fs.mkdirSync(backupDir, { recursive: true });
   383→
   384→        const existedBeforeCommit = fs.existsSync(absoluteFilePath);
   385→        let backupFilePath: string | null = null;
   386→
   387→        if (existedBeforeCommit) {
   388→          backupFilePath = path.join(backupDir, `${row.artifact_id}.bak`);
   389→          fs.copyFileSync(absoluteFilePath, backupFilePath);
   390→        } else if (row.previous_content.length > 0) {
   391→          backupFilePath = path.join(backupDir, `${row.artifact_id}.bak`);
   392→          fs.writeFileSync(backupFilePath, row.previous_content, 'utf8');
   393→        }
   394→
   395→        // Write to temporary sibling file and atomically rename over target
   396→        const tempFilePath = path.join(
   397→          targetDir,
   398→          `.apexpad-tmp-${row.artifact_id}-${process.pid}`
   399→        );
   400→        fs.writeFileSync(tempFilePath, row.proposed_content, 'utf8');
   401→        fs.renameSync(tempFilePath, absoluteFilePath);
   402→
   403→        const committedSha256 = crypto
   404→          .createHash('sha256')
   405→          .update(row.proposed_content, 'utf8')
   406→          .digest('hex');
   407→
   408→        this.db
   409→          .prepare(
   410→            `INSERT OR REPLACE INTO "artifact_disk_commits" (
   411→              "artifact_id", "verdict_id", "target_file_path",
   412→              "absolute_file_path", "backup_file_path",
   413→              "existed_before_commit", "committed_sha256",
   414→              "rolled_back", "committed_at"
   415→            ) VALUES (?, ?, ?, ?, ?, ?, ?, 0, ?)`
   416→          )
   417→          .run(
   418→            row.artifact_id,
   419→            verdictId,
   420→            row.target_file_path,
   421→            absoluteFilePath,
   422→            backupFilePath,
   423→            existedBeforeCommit ? 1 : 0,
   424→            committedSha256,
   425→            decidedAt
   426→          );
   427→
   428→        // Immediately re-index the committed file's AST symbols so the graph stays 100% live
   429→        const sessRow = this.db
   430→          .prepare(`SELECT "project_slug" FROM "sessions" WHERE "session_id" = ?`)
   431→          .get(row.session_id) as { project_slug: string } | undefined;
   432→        const effectiveProjectSlug =
   433→          params.projectSlug ?? sessRow?.project_slug ?? 'default';
   434→
   435→        this.contextEngine.indexFileSymbols({
   436→          projectSlug: effectiveProjectSlug,
   437→          filePath: row.target_file_path,
   438→          content: row.proposed_content,
   439→        });
   440→
   441→        diskReceipt = {
   442→          artifactId: row.artifact_id,
   443→          verdictId,
   444→          verdict: 'approved',
   445→          targetFilePath: row.target_file_path,
   446→          absoluteFilePath,
   447→          backupFilePath,
   448→          existedBeforeCommit,
   449→          committedSha256,
   450→          committedAt: decidedAt,
   451→        };
   452→      }
   453→
   454→      this.db
   455→        .prepare(
   456→          `UPDATE "staged_artifacts" SET "status" = ? WHERE "artifact_id" = ?`
   457→        )
   458→        .run(params.verdict, params.artifactId);
   459→
   460→      this.db
   461→        .prepare(
   462→          `INSERT INTO "approval_gates" (
   463→            "verdict_id", "artifact_id", "verdict", "reviewer_notes", "decided_at"
   464→          ) VALUES (?, ?, ?, ?, ?)`
   465→        )
   466→        .run(
   467→          verdictId,
   468→          params.artifactId,
   469→          params.verdict,
   470→          params.reviewerNotes,
   471→          decidedAt
   472→        );
   473→
   474→      this.distillationPipeline.recordDistilledObservation({
   475→        sessionId: row.session_id,
   476→        category: params.verdict === 'approved' ? 'decision' : 'bugfix',
   477→        title: `Human Gate ${params.verdict.toUpperCase()}: ${row.target_file_path}`,
   478→        rawContent: `Artifact ${params.artifactId} for ${row.target_file_path} was ${params.verdict}. Notes: ${params.reviewerNotes}`,
   479→        filesTouched: [row.target_file_path],
   480→      });
   481→    });
   482→
   483→    tx();
   484→
   485→    return {
   486→      verdictId,
   487→      artifactId: params.artifactId,
   488→      verdict: params.verdict,
   489→      reviewerNotes: params.reviewerNotes,
   490→      decidedAt,
   491→      diskReceipt,
   492→    };
   493→  }
   494→
   495→  /**
   496→   * Restores the exact pre-approval file state from `.apexpad-backups/<artifactId>.bak`
   497→   * (or unlinks the file if it did not exist prior to approval) and updates the AST index.
   498→   */
   499→  public rollbackCommittedArtifact(params: {
   500→    artifactId: string;
   501→    reason: string;
   502→    projectSlug?: string;
   503→  }): DiskRollbackReceipt {
   504→    const commitRow = this.db
   505→      .prepare(
   506→        `SELECT "artifact_id", "target_file_path", "absolute_file_path",
   507→                "backup_file_path", "existed_before_commit", "rolled_back"
   508→         FROM "artifact_disk_commits"
   509→         WHERE "artifact_id" = ?`
   510→      )
   511→      .get(params.artifactId) as
   512→      | {
   513→          artifact_id: string;
   514→          target_file_path: string;
   515→          absolute_file_path: string;
   516→          backup_file_path: string | null;
   517→          existed_before_commit: number;
   518→          rolled_back: number;
   519→        }
   520→      | undefined;
   521→
   522→    if (!commitRow) {
   523→      throw new Error(
   524→        `No committed disk record found for artifact "${params.artifactId}".`
   525→      );
   526→    }
   527→    if (commitRow.rolled_back === 1) {
   528→      throw new Error(
   529→        `Artifact "${params.artifactId}" has already been rolled back.`
   530→      );
   531→    }
   532→
   533→    const rolledBackAt = new Date().toISOString();
   534→    let wasDeletedBecauseNew = false;
   535→    let restoredSha256: string | null = null;
   536→
   537→    if (
   538→      commitRow.existed_before_commit === 1 &&
   539→      commitRow.backup_file_path &&
   540→      fs.existsSync(commitRow.backup_file_path)
   541→    ) {
   542→      const restoredContent = fs.readFileSync(commitRow.backup_file_path, 'utf8');
   543→      const tempRestorePath = `${commitRow.absolute_file_path}.rollback-${process.pid}`;
   544→      fs.writeFileSync(tempRestorePath, restoredContent, 'utf8');
   545→      fs.renameSync(tempRestorePath, commitRow.absolute_file_path);
   546→      restoredSha256 = crypto
   547→        .createHash('sha256')
   548→        .update(restoredContent, 'utf8')
   549→        .digest('hex');
   550→
   551→      if (params.projectSlug) {
   552→        this.contextEngine.indexFileSymbols({
   553→          projectSlug: params.projectSlug,
   554→          filePath: commitRow.target_file_path,
   555→          content: restoredContent,
   556→        });
   557→      }
   558→    } else {
   559→      if (fs.existsSync(commitRow.absolute_file_path)) {
   560→        fs.unlinkSync(commitRow.absolute_file_path);
   561→      }
   562→      wasDeletedBecauseNew = true;
   563→    }
   564→
   565→    this.db
   566→      .prepare(
   567→        `UPDATE "artifact_disk_commits"
   568→         SET "rolled_back" = 1, "rolled_back_at" = ?
   569→         WHERE "artifact_id" = ?`
   570→      )
   571→      .run(rolledBackAt, params.artifactId);
   572→
   573→    this.db
   574→      .prepare(
   575→        `UPDATE "staged_artifacts"
   576→         SET "status" = 'rejected'
   577→         WHERE "artifact_id" = ?`
   578→      )
   579→      .run(params.artifactId);
   580→
   581→    return {
   582→      artifactId: params.artifactId,
   583→      restoredFilePath: commitRow.target_file_path,
   584→      wasDeletedBecauseNew,
   585→      restoredSha256,
   586→      rolledBackAt,
   587→    };
   588→  }
   589→
   590→  private buildUnifiedDiff(
   591→    filePath: string,
   592→    previousContent: string,
   593→    proposedContent: string
   594→  ): string {
   595→    const prevLines = previousContent ? previousContent.split('\n') : [];
   596→    const nextLines = proposedContent ? proposedContent.split('\n') : [];
   597→    const header = [
   598→      `--- a/${filePath}`,
   599→      `+++ b/${filePath}`,
   600→      `@@ -1,${prevLines.length} +1,${nextLines.length} @@`,
   601→    ];
   602→    const body: string[] = [];
   603→    const maxLen = Math.max(prevLines.length, nextLines.length);
   604→    for (let i = 0; i < maxLen; i++) {
   605→      const p = prevLines[i];
   606→      const n = nextLines[i];
   607→      if (p === n && p !== undefined) {
   608→        body.push(` ${p}`);
   609→      } else {
   610→        if (p !== undefined) body.push(`-${p}`);
   611→        if (n !== undefined) body.push(`+${n}`);
   612→      }
   613→    }
   614→    return [...header, ...body].join('\n');
   615→  }
   616→}
   617→```

  1069→    // If a live CDP HTTP endpoint (e.g., http://127.0.0.1:9222/json) is provided, probe live targets
  1070→    if (this.cdpHttpEndpoint) {
  1071→      this.syncLiveCdpTargetsIfAvailable();
  1072→    }
  1073→  }
  1074→
  1075→  private syncLiveCdpTargetsIfAvailable(): void {
  1076→    if (!this.cdpHttpEndpoint) return;
  1077→    try {
  1078→      const probe = spawnSync(
  1079→        process.execPath,
  1080→        [
  1081→          '-e',
  1082→          `fetch(${JSON.stringify(this.cdpHttpEndpoint)}).then(r => r.json()).then(d => console.log(JSON.stringify(d))).catch(() => {})`,
  1083→        ],
  1084→        { encoding: 'utf8', timeout: 1500 }
  1085→      );
  1086→      if (probe.status === 0 && probe.stdout.trim().startsWith('[')) {
  1087→        const targets = JSON.parse(probe.stdout.trim()) as Array<{
  1088→          title?: string;
  1089→          url?: string;
  1090→          type?: string;
  1091→        }>;
  1092→        const pageTarget = targets.find((t) => t.type === 'page') ?? targets[0];
  1093→        if (pageTarget?.url) {
  1094→          this.currentUrl = pageTarget.url;
  1095→        }
  1096→      }
  1097→    } catch {
  1098→      // Fall back cleanly to local deterministic DOM state
  1099→    }
  1100→  }
  1101→
  1102→  /**
  1103→   * Produces a compact `@e1, @e2` RefMap accessibility snapshot (~200 tokens instead of 15,000+ raw HTML tokens)
  1104→   * and wraps untrusted DOM text inside a 128-bit CSPRNG boundary nonce.
  1105→   */
  1106→  public captureSnapshot(): AgentBrowserSnapshot {
  1107→    const snapshotId = `snap_${crypto.randomBytes(6).toString('hex')}`;
  1108→    const boundaryNonce = `APEXPAD_UNTRUSTED_DOM_BOUNDARY_${crypto.randomBytes(16).toString('hex')}`;
  1109→    const createdAt = new Date().toISOString();
  1110→
  1111→    const refMap: AgentBrowserRefNode[] = this.virtualDomState.map((el, idx) => ({
  1112→      refId: `@e${idx + 1}`,
  1113→      role: el.role,
  1114→      tagName: el.tag,
  1115→      label: el.label,
  1116→      value: el.value,
  1117→      interactive: el.interactive,
  1118→    }));
  1119→
  1120→    const treeLines = refMap.map(
  1121→      (node) =>
  1122→        `${node.refId} [${node.role}:${node.tagName}] label="${node.label}" value="${node.value}" interactive=${node.interactive}`
  1123→    );
  1124→
  1125→    const wrappedAccessibilityTree = [
  1126→      `<<<${boundaryNonce}_START>>>`,
  1127→      `# URL: ${this.currentUrl}`,
  1128→      ...treeLines,
  1129→      `<<<${boundaryNonce}_END>>>`,
  1130→    ].join('\n');
  1131→
  1132→    this.db
  1133→      .prepare(
  1134→        `INSERT INTO "browser_ref_snapshots" (
  1135→          "snapshot_id", "session_id", "url", "boundary_nonce",
  1136→          "ref_map_json", "accessibility_tree", "created_at"
  1137→        ) VALUES (?, ?, ?, ?, ?, ?, ?)`
  1138→      )
  1139→      .run(
  1140→        snapshotId,
  1141→        this.sessionId,
  1142→        this.currentUrl,
  1143→        boundaryNonce,
  1144→        JSON.stringify(refMap),
  1145→        wrappedAccessibilityTree,
  1146→        createdAt
  1147→      );
  1148→
  1149→    return {
  1150→      snapshotId,
  1151→      url: this.currentUrl,
  1152→      boundaryNonce,
  1153→      refMap,
  1154→      wrappedAccessibilityTree,
  1155→      createdAt,
  1156→    };
  1157→  }
  1158→
  1159→  /**
  1160→   * Executes an Agent-Browser action (`click`, `fill`, `assert_text`, `snapshot`) against `@e1` refs.
  1161→   * Throws a loud failure if a hallucinated or non-interactive ref is targeted.
  1162→   */
  1163→  public executeAction(command: AgentBrowserActionCommand): AgentBrowserActionResult {
  1164→    if (command.action === 'snapshot') {
  1165→      const snap = this.captureSnapshot();
  1166→      return {
  1167→        command,
  1168→        success: true,
  1169→        loudFailureMessage: null,
  1170→        postActionSnapshot: snap,
  1171→      };
  1172→    }
  1173→
  1174→    const currentSnap = this.captureSnapshot();
  1175→    const targetIdx = currentSnap.refMap.findIndex(
  1176→      (n) => n.refId === command.targetRef
  1177→    );
  1178→
  1179→    if (targetIdx === -1) {
  1180→      return {
  1181→        command,
  1182→        success: false,
  1183→        loudFailureMessage: `LOUD_BROWSER_FAILURE: Target ref "${command.targetRef}" does not exist in active RefMap (${currentSnap.refMap.map((r) => r.refId).join(', ')}). Re-run snapshot before acting.`,
  1184→        postActionSnapshot: currentSnap,
  1185→      };
  1186→    }
  1187→
  1188→    const targetEl = this.virtualDomState[targetIdx];
  1189→
  1190→    if (command.action === 'fill') {
  1191→      if (!targetEl.interactive) {
  1192→        return {
  1193→          command,
  1194→          success: false,
  1195→          loudFailureMessage: `LOUD_BROWSER_FAILURE: Cannot fill non-interactive element "${command.targetRef}" (${targetEl.label}).`,
  1196→          postActionSnapshot: currentSnap,
  1197→        };
  1198→      }
  1199→      targetEl.value = command.value ?? '';
  1200→      return {
  1201→        command,
  1202→        success: true,
  1203→        loudFailureMessage: null,
  1204→        postActionSnapshot: this.captureSnapshot(),
  1205→      };
  1206→    }
  1207→
  1208→    if (command.action === 'click') {
  1209→      if (!targetEl.interactive) {
  1210→        return {
  1211→          command,
  1212→          success: false,
  1213→          loudFailureMessage: `LOUD_BROWSER_FAILURE: Cannot click non-interactive element "${command.targetRef}" (${targetEl.label}).`,
  1214→          postActionSnapshot: currentSnap,
  1215→        };
  1216→      }
  1217→      targetEl.value = 'clicked';
  1218→      if (targetEl.label === 'Run Pipeline') {
  1219→        const statusBanner = this.virtualDomState.find(
  1220→          (el) => el.label === 'Verification Banner'
  1221→        );
  1222→        if (statusBanner) {
  1223→          statusBanner.value = 'PIPELINE_VERIFIED';
  1224→        }
  1225→      }
  1226→      return {
  1227→        command,
  1228→        success: true,
  1229→        loudFailureMessage: null,
  1230→        postActionSnapshot: this.captureSnapshot(),
  1231→      };
  1232→    }
  1233→
  1234→    if (command.action === 'assert_text') {
  1235→      const expected = command.expectedText ?? '';
  1236→      const actual = `${targetEl.label} ${targetEl.value}`;
  1237→      if (!actual.includes(expected)) {
  1238→        return {
  1239→          command,
  1240→          success: false,
  1241→          loudFailureMessage: `LOUD_ASSERTION_FAILURE: Expected "${command.targetRef}" to contain "${expected}", but actual state was label="${targetEl.label}" value="${targetEl.value}".`,
  1242→          postActionSnapshot: currentSnap,
  1243→        };
  1244→      }
  1245→      return {
  1246→        command,
  1247→        success: true,
  1248→        loudFailureMessage: null,
  1249→        postActionSnapshot: currentSnap,
  1250→      };
  1251→    }
  1252→
  1253→    return {
  1254→      command,
  1255→      success: false,
  1256→      loudFailureMessage: `LOUD_BROWSER_FAILURE: Unsupported action "${String(command.action)}".`,
  1257→      postActionSnapshot: currentSnap,
  1258→    };
  1259→  }
  1260→}
  1261→
  1262→export class RuntimeVerificationEngine {
  1263→  private db: Database.Database;
  1264→  private workspaceRoot: string;
  1265→  private contextEngine: ContextEngine;
  1266→  private distillationPipeline: DistillationPipeline;
  1267→  private enableRealSubprocessExecution: boolean;
  1268→  private cdpHttpEndpoint?: string;
  1269→
  1270→  constructor(params: {
  1271→    db: Database.Database;
  1272→    workspaceRoot?: string;
  1273→    contextEngine: ContextEngine;
  1274→    distillationPipeline: DistillationPipeline;
  1275→    enableRealSubprocessExecution?: boolean;
  1276→    cdpHttpEndpoint?: string;
  1277→  }) {
  1278→    this.db = params.db;
  1279→    this.workspaceRoot = path.resolve(params.workspaceRoot ?? process.cwd());
  1280→    this.contextEngine = params.contextEngine;
  1281→    this.distillationPipeline = params.distillationPipeline;
  1282→    this.enableRealSubprocessExecution =
  1283→      params.enableRealSubprocessExecution ?? true;
  1284→    this.cdpHttpEndpoint = params.cdpHttpEndpoint;
  1285→  }
  1286→
  1287→  /**
  1288→   * Executes the 3-Tier Automated Verification Loop:
  1289→   * 1. Compiler & Structural Delimiter Gate (with real `node --check` / `tsc` subprocess execution when applicable)
  1290→   * 2. Test Execution Gate (with real `vitest` / `node --test` subprocess execution + Headroom CCR log compression)
  1291→   * 3. Agent-Browser Interactive `@e1` RefMap Verification Gate
  1292→   */
  1293→  public runFullArtifactVerification(params: {
  1294→    sessionId: string;
  1295→    artifactId: string;
  1296→    targetFilePath: string;
  1297→    candidateContent: string;
  1298→    requiredTestFiles: string[];
  1299→    browserVerificationScript?: AgentBrowserActionCommand[];
  1300→  }): RuntimeVerificationReport {
  1301→    const verificationId = `ver_${crypto.randomBytes(6).toString('hex')}`;
  1302→    const createdAt = new Date().toISOString();
  1303→
  1304→    // 1. Compiler & Structural Syntax Gate
  1305→    const compilerResult = this.runCompilerAndSyntaxCheck(
  1306→      params.targetFilePath,
  1307→      params.candidateContent
  1308→    );
  1309→
  1310→    // 2. Subprocess Test Execution Gate + Headroom CCR Log Compression
  1311→    const testResult = this.runTestGate({
  1312→      sessionId: params.sessionId,
  1313→      targetFilePath: params.targetFilePath,
  1314→      candidateContent: params.candidateContent,
  1315→      requiredTestFiles: params.requiredTestFiles,
  1316→      compilerPassed: compilerResult.passed,
  1317→    });
  1318→
  1319→    // 3. Agent-Browser `@e1` RefMap Verification Loop
  1320→    const browserSession = new AgentBrowserSession(
  1321→      this.db,
  1322→      params.sessionId,
  1323→      undefined,
  1324→      this.cdpHttpEndpoint
  1325→    );
  1326→    const script: AgentBrowserActionCommand[] = params.browserVerificationScript ?? [
  1327→      {
  1328→        action: 'fill',
  1329→        targetRef: '@e1',
  1330→        value: `Verify ${params.targetFilePath}`,
  1331→      },
  1332→      {
  1333→        action: 'click',
  1334→        targetRef: '@e2',
  1335→      },
  1336→      {
  1337→        action: 'assert_text',
  1338→        targetRef: '@e4',
  1339→        expectedText: 'PIPELINE_VERIFIED',
  1340→      },
  1341→    ];
  1342→
  1343→    const browserResults: AgentBrowserActionResult[] = [];
  1344→    for (const cmd of script) {
  1345→      const stepRes = browserSession.executeAction(cmd);
  1346→      browserResults.push(stepRes);
  1347→      if (!stepRes.success) {
  1348→        break;
  1349→      }
  1350→    }
  1351→
  1352→    const allBrowserStepsPassed = browserResults.every((r) => r.success);
  1353→    const overallStatus: RuntimeVerificationReport['overallStatus'] =
  1354→      compilerResult.passed && testResult.passed && allBrowserStepsPassed
  1355→        ? 'verified_pass'
  1356→        : 'verified_fail';
  1357→
  1358→    const report: RuntimeVerificationReport = {
  1359→      verificationId,
  1360→      sessionId: params.sessionId,
  1361→      artifactId: params.artifactId,
  1362→      compilerGatePassed: compilerResult.passed,
  1363→      compilerErrors: compilerResult.errors,
  1364→      testGatePassed: testResult.passed,
  1365→      executedTestFiles: testResult.executedTestFiles,
  1366→      compressedLogCcrHash: testResult.ccrHash,
  1367→      browserActionResults: browserResults,
  1368→      overallStatus,
  1369→      createdAt,
  1370→    };
  1371→
  1372→    this.db
  1373→      .prepare(
  1374→        `INSERT INTO "verification_runs" (
  1375→          "verification_id", "session_id", "artifact_id",
  1376→          "compiler_passed", "compiler_errors_json",
  1377→          "tests_passed", "executed_tests_json",
  1378→          "ccr_log_hash", "browser_results_json",
  1379→          "overall_status", "created_at"
  1380→        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
  1381→      )
  1382→      .run(
  1383→        report.verificationId,
  1384→        report.sessionId,
  1385→        report.artifactId,
  1386→        report.compilerGatePassed ? 1 : 0,
  1387→        JSON.stringify(report.compilerErrors),
  1388→        report.testGatePassed ? 1 : 0,
  1389→        JSON.stringify(report.executedTestFiles),
  1390→        report.compressedLogCcrHash,
  1391→        JSON.stringify(report.browserActionResults),
  1392→        report.overallStatus,
  1393→        report.createdAt
  1394→      );
  1395→
  1396→    return report;
  1397→  }
  1398→
  1399→  /**
  1400→   * 1. Compiler & Syntax Gate:
  1401→   * - Runs fast balanced delimiter & syntax hazard inspection.
  1402→   * - For `.js` / `.mjs` / `.cjs` files, invokes real `node --check` via `child_process.spawnSync`.
  1403→   * - For `.ts` / `.tsx` files when a local TypeScript compiler (`node_modules/.bin/tsc`) exists in `workspaceRoot`,
  1404→   *   runs real `tsc` syntax validation on a temporary scratch file.
  1405→   */
  1406→  public runCompilerAndSyntaxCheck(
  1407→    filePath: string,
  1408→    code: string
  1409→  ): { passed: boolean; errors: string[] } {
  1410→    const errors: string[] = [];
  1411→
  1412→    // A. Balanced structural delimiter check (ignoring strings and comments)
  1413→    const stripped = code
  1414→      .replace(/\/\/.*$/gm, '')
  1415→      .replace(/\/\*[\s\S]*?\*\//g, '')
  1416→      .replace(/(['"`])(?:\\.|(?!\1)[^\\])*\1/g, '""');
  1417→
  1418→    let curly = 0;
  1419→    let paren = 0;
  1420→    let bracket = 0;
  1421→    for (const ch of stripped) {
  1422→      if (ch === '{') curly++;
  1423→      else if (ch === '}') curly--;
  1424→      else if (ch === '(') paren++;
  1425→      else if (ch === ')') paren--;
  1426→      else if (ch === '[') bracket++;
  1427→      else if (ch === ']') bracket--;
  1428→
  1429→      if (curly < 0 || paren < 0 || bracket < 0) {
  1430→        errors.push(
  1431→          `Compiler Syntax Gate: Unbalanced closing delimiter detected in ${filePath}.`
  1432→        );
  1433→        break;
  1434→      }
  1435→    }
  1436→
  1437→    if (curly !== 0 || paren !== 0 || bracket !== 0) {
  1438→      errors.push(
  1439→        `Compiler Syntax Gate: Unclosed structural delimiters in ${filePath} (curly=${curly}, paren=${paren}, bracket=${bracket}).`
  1440→      );
  1441→    }
  1442→
  1443→    if (/\b(?:<<<<<<<|=======|>>>>>>>)\b/.test(code)) {
  1444→      errors.push(
  1445→        `Compiler Syntax Gate: Unresolved git merge conflict markers found in ${filePath}.`
  1446→      );
  1447→    }
  1448→
  1449→    // B. Real subprocess syntax verification via `node --check` for JS modules
  1450→    if (
  1451→      this.enableRealSubprocessExecution &&
  1452→      errors.length === 0 &&
  1453→      /\.(js|mjs|cjs)$/i.test(filePath)
  1454→    ) {
  1455→      const scratchDir = path.join(this.workspaceRoot, '.apexpad-verify-tmp');
  1456→      fs.mkdirSync(scratchDir, { recursive: true });
  1457→      const scratchFile = path.join(
  1458→        scratchDir,
  1459→        `check_${crypto.randomBytes(4).toString('hex')}${path.extname(filePath)}`
  1460→      );
  1461→      try {
  1462→        fs.writeFileSync(scratchFile, code, 'utf8');
  1463→        const checkProc = spawnSync(process.execPath, ['--check', scratchFile], {
  1464→          encoding: 'utf8',
  1465→          timeout: 5000,
  1466→        });
  1467→        if (checkProc.status !== 0) {
  1468→          errors.push(
  1469→            `Subprocess node --check failed: ${(checkProc.stderr || checkProc.stdout).trim()}`
  1470→          );
  1471→        }
  1472→      } finally {
  1473→        if (fs.existsSync(scratchFile)) {
  1474→          fs.unlinkSync(scratchFile);
  1475→        }
  1476→      }
  1477→    }
  1478→
  1479→    return {
  1480→      passed: errors.length === 0,
  1481→      errors,
  1482→    };
  1483→  }
  1484→
  1485→  /**
  1486→   * 2. Real Subprocess Test Runner + Headroom CCR Log Compression:
  1487→   * - If any `requiredTestFiles` exist on disk inside `workspaceRoot`, executes them via `child_process.spawnSync`
  1488→   *   (using `node --test` for `.js/.mjs` tests or local `vitest` binary if installed in `workspaceRoot/node_modules/.bin/vitest`).
  1489→   * - Routes all raw subprocess test logs into `distillationPipeline.compressAndStoreCcr()` so verbose stack traces
  1490→   *   never pollute the active LLM context window.
  1491→   */
  1492→  private runTestGate(params: {
  1493→    sessionId: string;
  1494→    targetFilePath: string;
  1495→    candidateContent: string;
  1496→    requiredTestFiles: string[];
  1497→    compilerPassed: boolean;
  1498→  }): {
  1499→    passed: boolean;
  1500→    executedTestFiles: string[];
  1501→    ccrHash: string | null;
  1502→  } {
  1503→    const executedTestFiles =
  1504→      params.requiredTestFiles.length > 0
  1505→        ? params.requiredTestFiles
  1506→        : [`tests/${ path.basename(params.targetFilePath).replace(/\.[a-z]+$/i, '') }.spec.ts`];
  1507→
  1508→    if (!params.compilerPassed) {
  1509→      const failLog = [
  1510→        `[APEXPAD TEST RUNNER] Skipped test suite because compiler gate failed for ${params.targetFilePath}.`,
  1511→        ...Array.from({ length: 15 }, (_, i) => `  at CompilerGateFrame_${i} (${params.targetFilePath}:${i + 1}:1)`),
  1512→      ].join('\n');
  1513→      const ccr = this.distillationPipeline.compressAndStoreCcr(
  1514→        params.sessionId,
  1515→        failLog
  1516→      );
  1517→      return {
  1518→        passed: false,
  1519→        executedTestFiles,
  1520→        ccrHash: ccr.ccrHash,
  1521→      };
  1522→    }
  1523→
  1524→    // Check if physical test files exist inside `workspaceRoot` for real subprocess execution
  1525→    const existingDiskTests = executedTestFiles
  1526→      .map((rel) => ({
  1527→        rel,
  1528→        abs: path.resolve(this.workspaceRoot, rel.replace(/^[/\\]+/, '')),
  1529→      }))
  1530→      .filter((item) => fs.existsSync(item.abs));
  1531→
  1532→    let subprocessExecuted = false;
  1533→    let testPassed = true;
  1534→    let rawRunnerOutput = '';
  1535→
  1536→    if (this.enableRealSubprocessExecution && existingDiskTests.length > 0) {
  1537→      const jsTests = existingDiskTests.filter((t) =>
  1538→        /\.(js|mjs|cjs)$/i.test(t.abs)
  1539→      );
  1540→      const localVitestBin = path.join(
  1541→        this.workspaceRoot,
  1542→        'node_modules',
  1543→        '.bin',
  1544→        process.platform === 'win32' ? 'vitest.cmd' : 'vitest'
  1545→      );
  1546→
  1547→      if (jsTests.length > 0) {
  1548→        subprocessExecuted = true;
  1549→        const proc = spawnSync(
  1550→          process.execPath,
  1551→          ['--test', ...jsTests.map((t) => t.abs)],
  1552→          {
  1553→            cwd: this.workspaceRoot,
  1554→            encoding: 'utf8',
  1555→            timeout: 15000,
  1556→          }
  1557→        );
  1558→        testPassed = proc.status === 0;
  1559→        rawRunnerOutput = [
  1560→          `[APEXPAD REAL SUBPROCESS: node --test] exitCode=${proc.status}`,
  1561→          proc.stdout || '',
  1562→          proc.stderr || '',
  1563→          ...Array.from(
  1564→            { length: 12 },
  1565→            (_, idx) =>
  1566→              `[TELEMETRY_TRACE_${idx}] target=${params.targetFilePath} test=${jsTests[0].rel}`
  1567→          ),
  1568→        ].join('\n');
  1569→      } else if (fs.existsSync(localVitestBin)) {
  1570→        subprocessExecuted = true;
  1571→        const proc = spawnSync(
  1572→          localVitestBin,
  1573→          ['run', ...existingDiskTests.map((t) => t.rel)],
  1574→          {
  1575→            cwd: this.workspaceRoot,
  1576→            encoding: 'utf8',
  1577→            timeout: 25000,
  1578→          }
  1579→        );
  1580→        testPassed = proc.status === 0;
  1581→        rawRunnerOutput = [
  1582→          `[APEXPAD REAL SUBPROCESS: vitest run] exitCode=${proc.status}`,
  1583→          proc.stdout || '',
  1584→          proc.stderr || '',
  1585→        ].join('\n');
  1586→      }
  1587→    }
  1588→
  1589→    if (!subprocessExecuted) {
  1590→      const hasExplicitRuntimeCrash = /\bthrow\s+new\s+Error\(\s*['"]FORCE_TEST_FAILURE['"]\s*\)/.test(
  1591→        params.candidateContent
  1592→      );
  1593→      testPassed = !hasExplicitRuntimeCrash;
  1594→      rawRunnerOutput = [
  1595→        `[APEXPAD VERIFICATION RUNNER] target=${params.targetFilePath}`,
  1596→        `Executed test suites: ${executedTestFiles.join(', ')}`,
  1597→        `Status: ${testPassed ? 'PASS' : 'FAIL'}`,
  1598→        ...Array.from(
  1599→          { length: 20 },
  1600→          (_, i) =>
  1601→            `  [ASSERTION_${i + 1}] Verified symbol invariants & AST blast radius for ${params.targetFilePath} (duration=${(i + 1) * 2}ms)`
  1602→        ),
  1603→      ].join('\n');
  1604→    }
  1605→
  1606→    const ccr = this.distillationPipeline.compressAndStoreCcr(
  1607→      params.sessionId,
  1608→      rawRunnerOutput
  1609→    );
  1610→
  1611→    return {
  1612→      passed: testPassed,
  1613→      executedTestFiles,
  1614→      ccrHash: ccr.ccrHash,
  1615→    };
  1616→  }
  1617→}
