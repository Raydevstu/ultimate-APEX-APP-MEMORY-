    90→function EventFeed({ events }: { events: UnattendedCaptureEvent[] }) {
    91→  const [, setTick] = React.useState(0)
    92→
    93→  // Re-render periodically so relative timestamps stay fresh.
    94→  React.useEffect(() => {
    95→    const timer = setInterval(() => setTick((t) => t + 1), 15000)
    96→    return () => clearInterval(timer)
    97→  }, [])
    98→
    99→  return (
   100→    <div>
   101→      <TinyLabel className="mb-1.5">Live Event Feed (SSE)</TinyLabel>
   102→      <div
   103→        className={cn(
   104→          'max-h-64 space-y-1 overflow-y-auto rounded-md border border-wire bg-vault p-1.5',
   105→          THIN_SCROLL,
   106→        )}
   107→      >
   108→        {events.length === 0 ? (
   109→          <div className="p-3 text-center text-[11px] text-ink-muted">
   110→            Listening for capture events…
   111→          </div>
   112→        ) : (
   113→          <AnimatePresence initial={false}>
   114→            {events.map((event, i) => {
   115→              const meta = EVENT_ICON[event.eventType] ?? {
   116→                icon: <Wrench className="h-3 w-3" />,
   117→                color: 'text-ink-muted',
   118→              }
   119→              return (
   120→                <SlideInItem
   121→                  key={`${event.timestamp}-${event.eventType}-${i}`}
   122→                  className="flex items-start gap-2 rounded px-1.5 py-1 hover:bg-vault-1"
   123→                >
   124→                  <span className={cn('mt-0.5 shrink-0', meta.color)}>
   125→                    {meta.icon}
   126→                  </span>
   127→                  <div className="min-w-0 flex-1">
   128→                    <div className="flex items-baseline justify-between gap-2">
   129→                      <span className={cn('font-mono text-[10px] font-semibold', meta.color)}>
   130→                        {event.eventType}
   131→                      </span>
   132→                      <span className="shrink-0 font-mono text-[9px] text-ink-muted">
   133→                        {formatRelativeTime(event.timestamp)}
   134→                      </span>
   135→                    </div>
   136→                    <p className="break-all font-mono text-[9px] leading-relaxed text-ink-muted">
   137→                      {truncate(JSON.stringify(event.payload ?? {}), 120)}
   138→                    </p>
   139→                  </div>
   140→                </SlideInItem>
   141→              )
   142→            })}
   143→          </AnimatePresence>
   144→        )}
   145→      </div>
   146→    </div>
   147→  )

   862→            </Panel>
   863→          </FadeIn>
