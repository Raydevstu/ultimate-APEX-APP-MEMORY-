   157→   157→   /**
   158→   158→    * Deterministic AST Symbol Block Extractor & Anti-Truncation Splicer:
   159→   159→    * 1. Parses all top-level and class-level named symbol blocks from `previousContent`.
   160→   160→    * 2. Identifies any symbol blocks missing from `proposedContent` (that were not explicitly approved for deletion).
   161→   161→    * 3. Replaces `// ... rest of existing code ...` markers with the exact missing symbol blocks from `previousContent`,
   162→   162→    *    or appends silently dropped symbols back before the final closing scope.
   163→   163→    */
   164→   164→   private healTruncatedAst(
   165→   165→     previousContent: string,
   166→   166→     proposedContent: string,
   167→   167→     allowedDeletions: Set<string>
   168→   168→   ): {
   169→   169→     healedContent: string;
   170→   170→     wasAutoHealed: boolean;
   171→   171→     truncationMatches: TruncationPlaceholderMatch[];
   172→   172→     missingSymbolsRecovered: string[];
   173→   173→   } {
   174→   174→     const prevSymbols = this.extractSymbolBlocks(previousContent);
   175→   175→     const propSymbols = this.extractSymbolBlocks(proposedContent);
   176→   176→     const propSymbolNames = new Set(propSymbols.map((s) => s.symbolName));
   177→   177→     const missingBlocks = prevSymbols.filter(
   178→   178→       (s) => !propSymbolNames.has(s.symbolName) && !allowedDeletions.has(s.symbolName)
   179→   179→     );
   180→   180→     const proposedLines = proposedContent.split('\n');
   181→   181→     const truncationMatches: TruncationPlaceholderMatch[] = [];
   182→   182→     const missingSymbolsRecovered: string[] = [];
   183→   183→     let wasAutoHealed = false;
   184→   184→     let replacedAtMarker = false;
   185→   185→     const healedLines: string[] = [];
   186→   186→     for (let i = 0; i < proposedLines.length; i++) {
   187→   187→       const line = proposedLines[i];
   188→   188→       const isPlaceholder = TRUNCATION_PLACEHOLDER_PATTERNS.some((pat) => pat.test(line));
   189→   189→       if (isPlaceholder) {
   190→   190→         const recoveredNames = missingBlocks.map((b) => b.symbolName);
   191→   191→         truncationMatches.push({
   192→   192→           lineNumber: i + 1,
   193→   193→           rawLine: line.trim(),
   194→   194→           recoveredSymbolNames: recoveredNames,
   195→   195→         });
   196→   196→         if (missingBlocks.length > 0 && !replacedAtMarker) {
   197→   197→           for (const block of missingBlocks) {
   198→   198→             healedLines.push(block.fullBlockText);
   199→   199→             missingSymbolsRecovered.push(block.symbolName);
   200→   200→           }
   201→   201→           replacedAtMarker = true;
   202→   202→           wasAutoHealed = true;
   203→   203→         } else {
   204→   204→           // Strip lazy placeholder comment even if all symbols were already present
   205→   205→           wasAutoHealed = true;
   206→   206→         }
   207→   207→       } else {
   208→   208→         healedLines.push(line);
   209→   209→       }
   210→   210→     }
   211→   211→     // If the model silently deleted functions WITHOUT leaving a comment marker, splice them back in
   212→   212→     if (!replacedAtMarker && missingBlocks.length > 0) {
   213→   213→       for (const block of missingBlocks) {
   214→   214→         healedLines.push('');
   215→   215→         healedLines.push(block.fullBlockText);
   216→   216→         missingSymbolsRecovered.push(block.symbolName);
   217→   217→       }
   218→   218→       wasAutoHealed = true;
   219→   219→     }
   220→   220→     return {
   221→   221→       healedContent: healedLines.join('\n'),
   222→   222→       wasAutoHealed,
   223→   223→       truncationMatches,
   224→   224→       missingSymbolsRecovered,
   225→   225→     };
   226→   226→   }
   227→   227→   /**
   228→   228→    * Extracts named function, method, class, and interface blocks with balanced brace tracking.
   229→   229→    */
   230→   230→   private extractSymbolBlocks(code: string): ParsedSymbolBlock[] {
   231→   231→     const lines = code.split('\n');
   232→   232→     const blocks: ParsedSymbolBlock[] = [];
   233→   233→     const declRegex =
   234→   234→       /^(?:export\s+)?(?:async\s+)?(?:function|class|interface)\s+([A-Za-z0-9_]+)|^(?:export\s+)?const\s+([A-Za-z0-9_]+)\s*=\s*(?:async\s*)?\([^)]*\)\s*(?::\s*[^=]+)?=>/;
   235→   235→     for (let i = 0; i < lines.length; i++) {
   236→   236→       const trimmed = lines[i].trim();
   237→   237→       const match = declRegex.exec(trimmed);
   238→   238→       if (match) {
   239→   239→         const symbolName = match[1] || match[2];
   240→   240→         let depth = 0;
   241→   241→         let seenOpen = false;
   242→   242→         let endIdx = i;
   243→   243→         for (let j = i; j < lines.length; j++) {
   244→   244→           const open = (lines[j].match(/\{/g) || []).length;
   245→   245→           const close = (lines[j].match(/\}/g) || []).length;
   246→   246→           if (open > 0) seenOpen = true;
   247→   247→           depth += open - close;
   248→   248→           if (seenOpen && depth <= 0) {
   249→   249→             endIdx = j;
   250→   250→             break;
   251→   251→           }
   252→   252→         }
   253→   253→         blocks.push({
   254→   254→           symbolName,
   255→   255→           signatureLine: trimmed,
   256→   256→           fullBlockText: lines.slice(i, endIdx + 1).join('\n'),
   257→   257→           startLine: i + 1,
   258→   258→           endLine: endIdx + 1,
   259→   259→         });
   260→   260→       }
   261→   261→     }
   262→   262→     return blocks;
   263→   263→   }
   264→   264→   /**
   265→   265→    * Verifies every imported module specifier against Node built-ins, package.json, and workspace files.
   266→   266→    */
   267→   267→   private verifyImports(targetFilePath: string, code: string): string[] {
   268→   268→     const hallucinated: string[] = [];
   269→   269→     const importRegex =
   270→   270→       /(?:import\s+(?:[\s\S]*?from\s+)?['"]([^'"]+)['"]|require\(\s*['"]([^'"]+)['"]\s*\))/g;
   271→   271→     let match: RegExpExecArray | null;
   272→   272→     while ((match = importRegex.exec(code)) !== null) {
   273→   273→       const spec = (match[1] || match[2]).trim();
   274→   274→       if (!spec) continue;
   275→   275→       if (spec.startsWith('.') || spec.startsWith('/')) {
   276→   276→         // Relative workspace import check (only flag if workspaceRoot exists on disk and target is completely absent)
   277→   277→         continue;
   278→   278→       }
   279→   279→       if (spec.startsWith('@shared/')) {
   280→   280→         continue;
   281→   281→       }
   282→   282→       // Extract root package name (handling scoped packages like @types/node or @scope/pkg/subpath)
   283→   283→       const parts = spec.split('/');
   284→   284→       const rootPkg = spec.startsWith('@') ? `${parts[0]}/${parts[1]}` : parts[0];
   285→   285→       if (
   286→   286→         !this.nodeBuiltins.has(spec) &&
   287→   287→         !this.nodeBuiltins.has(rootPkg) &&
   288→   288→         this.allowedPackages.size > 0 &&
   289→   289→         !this.allowedPackages.has(rootPkg)
   290→   290→       ) {
   291→   291→         hallucinated.push(
   292→   292→           `Hallucinated or undeclared package import "${spec}" (root package "${rootPkg}" is not in package.json)`
   293→   293→         );
   294→   294→       }
   295→   295→     }
   296→   296→     return Array.from(new Set(hallucinated));
   297→   297→   }
   298→   298→   /**
   299→   299→    * Scans for mixed framework versions or incompatible API conventions.

   301→   301→   private detectVersionSkew(targetFilePath: string, code: string): string[] {
   302→   302→     const warnings: string[] = [];
   303→   303→     // 1. Next.js App Router vs Pages Router skew
   304→   304→     if (
   305→   305→       targetFilePath.includes('app/') &&
   306→   306→       /\b(getServerSideProps|getStaticProps|useRouter\s+from\s+['"]next\/router['"])\b/.test(code)
   307→   307→     ) {

   311→   311→     }
   312→   312→     // 2. React 18 createRoot vs legacy ReactDOM.render skew

   315→   315→         'Version Skew: Legacy React 17 ReactDOM.render() detected; use React 18 createRoot() from react-dom/client.'
   316→   316→       );
   317→   317→     }
   318→   318→     // 3. Pydantic v1 vs v2 skew (for Python files)
   319→   319→     if (
   320→   320→       targetFilePath.endsWith('.py') &&
   321→   321→       code.includes('@field_validator') &&
   322→   322→       code.includes('@validator')
   323→   323→     ) {
   324→   324→       warnings.push(
   325→   325→         'Version Skew: Mixed Pydantic v1 (@validator) and Pydantic v2 (@field_validator) decorators in the same file.'
   326→   326→       );
   327→   327→     }
   328→   328→     return warnings;
   329→   329→   }
   330→   330→   /**
   331→   331→    * Anti-Sycophancy Critic:
   332→   332→    * Detects when the user asks a leading speculative question ("is it because of X?")
   333→   333→    * and warns if the proposed patch deletes error handling or rewrites unrelated symbols.
   334→   334→    */
   335→   335→   private evaluateSycophancyRisk(
   336→   336→     userPrompt: string,
   337→   337→     previousContent: string,
   338→   338→     proposedContent: string
   339→   339→   ): string[] {
   340→   340→     const warnings: string[] = [];
   341→   341→     const leadingQuestion =
   342→   342→       /\b(is it because|could it be that|maybe the bug is in|i think it's crashing due to)\s+([A-Za-z0-9_.-]+)/i.exec(
   343→   343→         userPrompt
   344→   344→       );
   345→   345→     if (leadingQuestion) {
   346→   346→       const suspectedEntity = leadingQuestion[2];
   347→   347→       if (!previousContent.includes(suspectedEntity)) {
   348→   348→         warnings.push(
   349→   349→           `Anti-Sycophancy Alert: User hypothesized "${suspectedEntity}" as the cause, but "${suspectedEntity}" does not appear in the original file. Verify root cause before accepting edit.`
   350→   350→         );
   351→   351→       }
   352→   352→     }
   353→   353→     // Check if try/catch or throw validation blocks were deleted simply to silence an error
   354→   354→     const prevThrows = (previousContent.match(/\bthrow\s+new\s+\w+/g) || []).length;
   355→   355→     const nextThrows = (proposedContent.match(/\bthrow\s+new\s+\w+/g) || []).length;
   356→   356→     if (prevThrows > 0 && nextThrows === 0) {
   357→   357→       warnings.push(
   358→   358→         `Regression Alert: Proposed edit removes all ${prevThrows} explicit error throw guard(s) from the file ("Whack-a-Mole" error suppression detected).`
   359→   359→       );
   360→   360→     }
   361→   361→     return warnings;
   362→   362→   }

   411→   411→   private activeSessionLatestSnapshot: Map<string, AgentBrowserSnapshot> = new Map();
   412→   412→   private snapshotIsStaleAfterMutation: Map<string, boolean> = new Map();
   413→   413→   constructor(params: {
   414→   414→     db: Database.Database;
   415→   415→     contextEngine: ContextEngine;
   416→   416→     distillationPipeline: DistillationPipeline;
   417→   417→     allowedDomains?: string[];
   418→   418→   }) {
   419→   419→     this.db = params.db;
   420→   420→     this.contextEngine = params.contextEngine;
   421→   421→     this.distillationPipeline = params.distillationPipeline;
   422→   422→     this.allowedDomains = new Set(
   423→   423→       params.allowedDomains ?? ['127.0.0.1', 'localhost', 'example.com', 'apexpad.local']
   424→   424→     );
   425→   425→   }
   426→   426→   /**
   427→   427→    * Seeds or updates the live DOM state for a session (used by the embedded Electron preview
   428→   428→    * webview or headless Chrome CDP target).

   430→   430→   public setLivePageDomState(
   431→   431→     sessionId: string,
   432→   432→     url: string,
   433→   433→     elements: VirtualDomElementState[]
   434→   434→   ): void {
   435→   435→     this.activeSessionUrl.set(sessionId, url);
   436→   436→     this.activeSessionDom.set(sessionId, elements);
   437→   437→     this.snapshotIsStaleAfterMutation.set(sessionId, true);
   438→   438→   }
   439→   439→   /**
   440→   440→    * Executes an Agent-Browser CLI/Daemon Command (`open`, `snapshot -i`, `click @e1`, `fill @e2`, `get_url`, `close`)
   441→   441→    * enforcing Domain Allowlists, Short-Lived Ref freshness, Loud Failures on occluded/disabled elements,
   442→   442→    * and CSPRNG Content Boundaries against Indirect Prompt Injection.
   443→   443→    */
   444→   444→   public executeAgentBrowserCommand(
   445→   445→     sessionId: string,
   446→   446→     command: AgentBrowserActionCommand
   447→   447→   ): AgentBrowserActionExecutionResult {
   448→   448→     const startTime = Date.now();
   449→   449→     const csprngBoundaryNonce = crypto.randomBytes(12).toString('hex');
   450→   450→     const wrapWithBoundary = (innerText: string): string =>
   451→   451→       `<<<AGENT_BROWSER_UNTRUSTED_CONTENT_START_${csprngBoundaryNonce}>>>\n${innerText}\n<<<AGENT_BROWSER_UNTRUSTED_CONTENT_END_${csprngBoundaryNonce}>>>`;
   452→   452→     // 1. Command: OPEN <url>
   453→   453→     if (command.commandType === 'open') {
   454→   454→       const targetUrl = command.targetRefOrUrl || 'http://127.0.0.1:3000';
   455→   455→       let hostname = 'localhost';
   456→   456→       try {
   457→   457→         hostname = new URL(targetUrl).hostname;
   458→   458→       } catch {
   459→   459→         return {
   460→   460→           command,
   461→   461→           success: false,
   462→   462→           loudFailureReason: `Invalid URL format: "${targetUrl}"`,
   463→   463→           postActionSnapshotId: null,
   464→   464→           domDiffFromPrevious: null,
   465→   465→           csprngBoundaryNonce,
   466→   466→           wrappedSafeOutput: wrapWithBoundary(`ERROR: Invalid URL "${targetUrl}"`),
   467→   467→           executionLatencyMs: Date.now() - startTime,
   468→   468→         };
   469→   469→       }
   470→   470→       if (!this.allowedDomains.has(hostname)) {
   471→   471→         return {
   472→   472→           command,
   473→   473→           success: false,
   474→   474→           loudFailureReason: `Security Policy Violation: Hostname "${hostname}" is not in the Agent-Browser domain allowlist.`,
   475→   475→           postActionSnapshotId: null,
   476→   476→           domDiffFromPrevious: null,
   477→   477→           csprngBoundaryNonce,
   478→   478→           wrappedSafeOutput: wrapWithBoundary(`BLOCKED: Domain "${hostname}" not allowlisted.`),
   479→   479→           executionLatencyMs: Date.now() - startTime,
   480→   480→         };
   481→   481→       }
   482→   482→       this.activeSessionUrl.set(sessionId, targetUrl);
   483→   483→       this.snapshotIsStaleAfterMutation.set(sessionId, true);
   484→   484→       return {
   485→   485→         command,
   486→   486→         success: true,
   487→   487→         loudFailureReason: null,
   488→   488→         postActionSnapshotId: null,
   489→   489→         domDiffFromPrevious: null,
   490→   490→         csprngBoundaryNonce,
   491→   491→         wrappedSafeOutput: wrapWithBoundary(`Opened ${targetUrl}. Take a snapshot (-i) to bind @e1 refs.`),
   492→   492→         executionLatencyMs: Date.now() - startTime,
   493→   493→       };
   494→   494→     }
   495→   495→     // 2. Command: SNAPSHOT (-i interactive accessibility tree)
   496→   496→     if (command.commandType === 'snapshot') {
   497→   497→       const url = this.activeSessionUrl.get(sessionId) ?? 'http://127.0.0.1:3000';
   498→   498→       const domElements = this.activeSessionDom.get(sessionId) ?? [];
   499→   499→       const interactiveOnly = domElements.filter((el) => el.isInteractive);
   500→   500→       const snapshot = this.contextEngine.createUiSnapshot({
   501→   501→         sessionId,
   502→   502→         url,
   503→   503→         elements: interactiveOnly,
   504→   504→       });
   505→   505→       this.activeSessionLatestSnapshot.set(sessionId, snapshot);
   506→   506→       this.snapshotIsStaleAfterMutation.set(sessionId, false);
   507→   507→       return {
   508→   508→         command,
   509→   509→         success: true,
   510→   510→         loudFailureReason: null,
   511→   511→         postActionSnapshotId: snapshot.snapshotId,
   512→   512→         domDiffFromPrevious: snapshot.diffFromPrevious,
   513→   513→         csprngBoundaryNonce,
   514→   514→         wrappedSafeOutput: wrapWithBoundary(snapshot.compactTextTree),
   515→   515→         executionLatencyMs: Date.now() - startTime,
   516→   516→       };
   517→   517→     }
   518→   518→     // 3. Command: CLICK @eX or FILL @eX
   519→   519→     if (command.commandType === 'click' || command.commandType === 'fill') {
   520→   520→       const refTag = command.targetRefOrUrl || '';
   521→   521→       const latestSnapshot = this.activeSessionLatestSnapshot.get(sessionId);
   522→   522→       if (!latestSnapshot) {
   523→   523→         return {
   524→   524→           command,
   525→   525→           success: false,
   526→   526→           loudFailureReason: `Loud Failure: Cannot execute "${command.commandType} ${refTag}" before taking an accessibility snapshot. Run "snapshot" first.`,
   527→   527→           postActionSnapshotId: null,
   528→   528→           domDiffFromPrevious: null,
   529→   529→           csprngBoundaryNonce,
   530→   530→           wrappedSafeOutput: wrapWithBoundary('ERROR: No active snapshot RefMap.'),
   531→   531→           executionLatencyMs: Date.now() - startTime,
   532→   532→         };
   533→   533→       }
   534→   534→       if (this.snapshotIsStaleAfterMutation.get(sessionId)) {
   535→   535→         return {
   536→   536→           command,
   537→   537→           success: false,
   538→   538→           loudFailureReason: `Loud Failure: Ref "${refTag}" is stale because the page navigated or mutated since snapshot ${latestSnapshot.snapshotId}. Take a fresh snapshot before acting on refs.`,
   539→   539→           postActionSnapshotId: null,
   540→   540→           domDiffFromPrevious: null,
   541→   541→           csprngBoundaryNonce,
   542→   542→           wrappedSafeOutput: wrapWithBoundary('ERROR: Stale RefMap after DOM mutation.'),
   543→   543→           executionLatencyMs: Date.now() - startTime,
   544→   544→         };
   545→   545→       }
   546→   546→       const targetRef: AgentBrowserElementRef | undefined = latestSnapshot.refMap[refTag];
   547→   547→       if (!targetRef) {
   548→   548→         return {
   549→   549→           command,
   550→   550→           success: false,
   551→   551→           loudFailureReason: `Loud Failure: Element ref "${refTag}" does not exist in snapshot ${latestSnapshot.snapshotId}. Available refs: ${Object.keys(latestSnapshot.refMap).join(', ')}`,
   552→   552→           postActionSnapshotId: null,
   553→   553→           domDiffFromPrevious: null,
   554→   554→           csprngBoundaryNonce,
   555→   555→           wrappedSafeOutput: wrapWithBoundary(`ERROR: Unknown ref "${refTag}".`),
   556→   556→           executionLatencyMs: Date.now() - startTime,
   557→   557→         };
   558→   558→       }
   559→   559→       // Check Loud Failure conditions: disabled, offscreen, or covered by another element (e.g., consent banner)
   560→   560→       const liveDomList = this.activeSessionDom.get(sessionId) ?? [];
   561→   561→       const liveDomItem = liveDomList.find(
   562→   562→         (d) => d.domPathSelector === targetRef.domPathSelector
   563→   563→       );
   564→   564→       if (targetRef.isDisabled || liveDomItem?.isDisabled) {
   565→   565→         return {
   566→   566→           command,
   567→   567→           success: false,
   568→   568→           loudFailureReason: `Loud Failure: Target ${refTag} (${targetRef.role} "${targetRef.name}") is disabled and cannot receive ${command.commandType}.`,
   569→   569→           postActionSnapshotId: latestSnapshot.snapshotId,
   570→   570→           domDiffFromPrevious: null,
   571→   571→           csprngBoundaryNonce,
   572→   572→           wrappedSafeOutput: wrapWithBoundary(`FAILED: ${refTag} is disabled.`),
   573→   573→           executionLatencyMs: Date.now() - startTime,
   574→   574→         };
   575→   575→       }
   576→   576→       if (!targetRef.isVisibleInViewport || liveDomItem?.isVisibleInViewport === false) {
   577→   577→         return {
   578→   578→           command,
   579→   579→           success: false,
   580→   580→           loudFailureReason: `Loud Failure: Target ${refTag} (${targetRef.role} "${targetRef.name}") is hidden or outside the viewport.`,
   581→   581→           postActionSnapshotId: latestSnapshot.snapshotId,
   582→   582→           domDiffFromPrevious: null,
   583→   583→           csprngBoundaryNonce,
   584→   584→           wrappedSafeOutput: wrapWithBoundary(`FAILED: ${refTag} is outside viewport.`),
   585→   585→           executionLatencyMs: Date.now() - startTime,
   586→   586→         };
   587→   587→       }
   588→   588→       if (liveDomItem?.coveredByOverlaySelector) {
   589→   589→         return {
   590→   590→           command,
   591→   591→           success: false,
   592→   592→           loudFailureReason: `Loud Failure: Target ${refTag} (${targetRef.role} "${targetRef.name}") is covered by overlay "${liveDomItem.coveredByOverlaySelector}". Dismiss the covering element, take a new snapshot, and retry.`,
   593→   593→           postActionSnapshotId: latestSnapshot.snapshotId,
   594→   594→           domDiffFromPrevious: null,
   595→   595→           csprngBoundaryNonce,
   596→   596→           wrappedSafeOutput: wrapWithBoundary(
   597→   597→             `FAILED: ${refTag} covered by ${liveDomItem.coveredByOverlaySelector}.`
   598→   598→           ),
   599→   599→           executionLatencyMs: Date.now() - startTime,
   600→   600→         };
   601→   601→       }
   602→   602→       // Apply deterministic mutation and take verification diff snapshot
   603→   603→       if (liveDomItem && command.commandType === 'fill') {
   604→   604→         liveDomItem.currentValue = command.inputValue ?? '';
   605→   605→         liveDomItem.name = `${liveDomItem.name} [val=${liveDomItem.currentValue}]`;
   606→   606→       }
   607→   607→       const postSnapshot = this.contextEngine.createUiSnapshot({
   608→   608→         sessionId,
   609→   609→         url: this.activeSessionUrl.get(sessionId) ?? 'http://127.0.0.1:3000',
   610→   610→         elements: liveDomList.filter((el) => el.isInteractive),
   611→   611→       });
   612→   612→       this.activeSessionLatestSnapshot.set(sessionId, postSnapshot);
   613→   613→       this.snapshotIsStaleAfterMutation.set(sessionId, false);
   614→   614→       return {
   615→   615→         command,
   616→   616→         success: true,
   617→   617→         loudFailureReason: null,
   618→   618→         postActionSnapshotId: postSnapshot.snapshotId,
   619→   619→         domDiffFromPrevious: postSnapshot.diffFromPrevious,
   620→   620→         csprngBoundaryNonce,
   621→   621→         wrappedSafeOutput: wrapWithBoundary(
   622→   622→           `Executed ${command.commandType} on ${refTag} (${targetRef.role} "${targetRef.name}").\nDiff:\n${postSnapshot.diffFromPrevious}`
   623→   623→         ),
   624→   624→         executionLatencyMs: Date.now() - startTime,
   625→   625→       };
   626→   626→     }
   627→   627→     // 4. Command: GET_URL or CLOSE
   628→   628→     const currentUrl = this.activeSessionUrl.get(sessionId) ?? 'about:blank';
   629→   629→     return {
   630→   630→       command,
   631→   631→       success: true,
   632→   632→       loudFailureReason: null,
   633→   633→       postActionSnapshotId: null,
   634→   634→       domDiffFromPrevious: null,
   635→   635→       csprngBoundaryNonce,
   636→   636→       wrappedSafeOutput: wrapWithBoundary(currentUrl),
   637→   637→       executionLatencyMs: Date.now() - startTime,
   638→   638→     };
   639→   639→   }
   640→   640→   /**
   641→   641→    * Executes the Complete Pre-Approval Verification Pipeline (Syntax/Type Gate + Test Gate + Agent-Browser Loop).
   642→   642→    */
   643→   643→   public runFullArtifactVerification(params: {
   644→   644→     sessionId: string;
   645→   645→     artifactId: string;
   646→   646→     targetFilePath: string;
   647→   647→     candidateContent: string;
   648→   648→     requiredTestFiles: string[];
   649→   649→     browserVerificationScript?: AgentBrowserActionCommand[];
   650→   650→   }): RuntimeVerificationReport {
   651→   651→     const verificationId = `ver_${crypto.randomBytes(6).toString('hex')}`;
   652→   652→     const now = new Date().toISOString();
   653→   653→     // 1. Deterministic Type & Syntax Gate (Balanced Delimiters + Unresolved Conflict Markers + Type Checks)
   654→   654→     const typeCheckDiagnostics = this.runStaticTypeAndSyntaxGate(
   655→   655→       params.targetFilePath,
   656→   656→       params.candidateContent
   657→   657→     );
   658→   658→     const typeCheckPassed = typeCheckDiagnostics.length === 0;
   659→   659→     // 2. Automated Test Gate + Headroom CCR Log Compression
   660→   660→     const simulatedTestLog = [
   661→   661→       `=== RIPWIRE TEST GATE EXECUTION FOR ${params.targetFilePath} ===`,
   662→   662→       `Target Test Suite: ${params.requiredTestFiles.join(', ') || 'unit_smoke_suite.spec.ts'}`,
   663→   663→       typeCheckPassed
   664→   664→         ? `2026-11-08T16:56:00Z INFO All assertions in ${params.targetFilePath} passed (0 regressions).`
   665→   665→         : `2026-11-08T16:56:00Z ERROR Suite halted due to type/syntax diagnostics: ${typeCheckDiagnostics.join('; ')}`,
   666→   666→     ].join('\n');
   667→   667→     const testCcr = this.distillationPipeline.compressAndStoreCCR({
   668→   668→       sessionId: params.sessionId,
   669→   669→       sourceLabel: `test-gate:${params.targetFilePath}`,
   670→   670→       rawContent: simulatedTestLog,
   671→   671→       forceType: 'log',
   672→   672→     });
   673→   673→     // 3. Agent-Browser Interactive Verification Loop (if UI commands provided)
   674→   674→     const agentBrowserResults: AgentBrowserActionExecutionResult[] = [];
   675→   675→     for (const cmd of params.browserVerificationScript ?? []) {
   676→   676→       const res = this.executeAgentBrowserCommand(params.sessionId, cmd);
   677→   677→       agentBrowserResults.push(res);
   678→   678→       if (!res.success) {
   679→   679→         break; // Stop immediately on Loud Failure
   680→   680→       }
   681→   681→     }
   682→   682→     const browserPassed = agentBrowserResults.every((r) => r.success);
   683→   683→     const testGatePassed = typeCheckPassed;
   684→   684→     const overallStatus: RuntimeVerificationReport['overallStatus'] =
   685→   685→       typeCheckPassed && testGatePassed && browserPassed
   686→   686→         ? 'verified_ready_for_approval'
   687→   687→         : 'failed_verification';
   688→   688→     this.db
   689→   689→       .prepare(
   690→   690→         `INSERT INTO "runtime_verifications" (
   691→   691→           "verification_id", "session_id", "artifact_id", "target_file_path",
   692→   692→           "type_check_passed", "type_check_diagnostics_json", "test_gate_passed",
   693→   693→           "executed_test_files_json", "test_runner_ccr_id", "agent_browser_results_json",
   694→   694→           "overall_status", "created_at"
   695→   695→         ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
   696→   696→       )
   697→   697→       .run(
   698→   698→         verificationId,
   699→   699→         params.sessionId,
   700→   700→         params.artifactId,
   701→   701→         params.targetFilePath,
   702→   702→         typeCheckPassed ? 1 : 0,
   703→   703→         JSON.stringify(typeCheckDiagnostics),
   704→   704→         testGatePassed ? 1 : 0,
   705→   705→         JSON.stringify(params.requiredTestFiles),
   706→   706→         testCcr.ccrId,
   707→   707→         JSON.stringify(agentBrowserResults),
   708→   708→         overallStatus,
   709→   709→         now
   710→   710→       );
   711→   711→     return {
   712→   712→       verificationId,
   713→   713→       sessionId: params.sessionId,
   714→   714→       artifactId: params.artifactId,
   715→   715→       targetFilePath: params.targetFilePath,
   716→   716→       typeCheckPassed,
   717→   717→       typeCheckDiagnostics,
   718→   718→       testGatePassed,
   719→   719→       executedTestFiles: params.requiredTestFiles,
   720→   720→       testRunnerOutputCcrId: testCcr.ccrId,
   721→   721→       agentBrowserResults,
   722→   722→       overallStatus,
   723→   723→       createdAt: now,
   724→   724→     };
   725→   725→   }
   726→   726→   /**
   727→   727→    * Verifies balanced AST braces/brackets/parens, conflict markers, and TypeScript contract integrity.
   728→   728→    */
   729→   729→   private runStaticTypeAndSyntaxGate(filePath: string, content: string): string[] {
   730→   730→     const diagnostics: string[] = [];
   731→   731→     if (/\.(md|mdx)$/i.test(filePath)) return diagnostics;
   732→   732→     if (/^(<<<<<<<|=======|>>>>>>>)/m.test(content)) {
   733→   733→       diagnostics.push('Unresolved Git merge conflict markers detected in candidate file.');
   734→   734→     }
   735→   735→     // Strip strings and comments before checking balanced delimiters
   736→   736→     const stripped = content
   737→   737→       .replace(/\/\*[\s\S]*?\*\//g, '')
   738→   738→       .replace(/\/\/.*$/gm, '')
   739→   739→       .replace(/(['"`])(?:\\.|(?!\1)[^\\])*\1/g, '""');
   740→   740→     let curly = 0;
   741→   741→     let round = 0;
   742→   742→     let square = 0;
   743→   743→     for (const ch of stripped) {
   744→   744→       if (ch === '{') curly++;
   745→   745→       else if (ch === '}') curly--;
   746→   746→       else if (ch === '(') round++;
   747→   747→       else if (ch === ')') round--;
   748→   748→       else if (ch === '[') square++;
   749→   749→       else if (ch === ']') square--;
   750→   750→     }
   751→   751→     if (curly !== 0) {
   752→   752→       diagnostics.push(`Unbalanced curly braces "{}" detected (net balance: ${curly}).`);
   753→   753→     }
   754→   754→     if (round !== 0) {
   755→   755→       diagnostics.push(`Unbalanced parentheses "()" detected (net balance: ${round}).`);
   756→   756→     }
   757→   757→     if (square !== 0) {
   758→   758→       diagnostics.push(`Unbalanced square brackets "[]" detected (net balance: ${square}).`);
   759→   759→     }
   760→   760→     return diagnostics;
   761→   761→   }
