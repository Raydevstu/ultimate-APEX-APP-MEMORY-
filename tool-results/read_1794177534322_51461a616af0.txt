    92→    92→
    93→    93→  // Re-render periodically so relative timestamps stay fresh.

    95→    95→    const timer = setInterval(() => setTick((t) => t + 1), 15000)

    98→    98→
    99→    99→  return (
   100→   100→    <div>
   101→   101→      <TinyLabel className="mb-1.5">Live Event Feed (SSE)</TinyLabel>

   105→   105→          THIN_SCROLL,
   106→   106→        )}
   107→   107→      >
   108→   108→        {events.length === 0 ? (
   109→   109→          <div className="p-3 text-center text-[11px] text-[#8b949e]">
   110→   110→            Listening for capture events…
   111→   111→          </div>
   112→   112→        ) : (
   113→   113→          <AnimatePresence initial={false}>
   114→   114→            {events.map((event, i) => {
