   511→function BrowserConsole({ sessionId }: { sessionId: string }) {
   512→  const { toast } = useToast()
   513→  const [url, setUrl] = React.useState(BROWSER_DEMO_URL)
   514→  const [elementsJson, setElementsJson] = React.useState(BROWSER_DEMO_ELEMENTS)
   515→  const [commandType, setCommandType] = React.useState<AgentBrowserActionCommand['commandType']>('snapshot')
   516→  const [targetRef, setTargetRef] = React.useState('@e2')
   517→  const [inputValue, setInputValue] = React.useState('')
   518→  const [busy, setBusy] = React.useState<string | null>(null)
   519→  const [log, setLog] = React.useState<BrowserLogEntry[]>([])
   520→  const [state, setState] = React.useState<{
   521→    livePageRevision: number
   522→    refIds: string[]
   523→    actionLog: string[]
   524→  } | null>(null)
   525→
   526→  const parseElements = (): unknown[] | null => {
   527→    try {
   528→      const parsed: unknown = JSON.parse(elementsJson)
   529→      if (!Array.isArray(parsed)) throw new Error('elements must be a JSON array')
   530→      return parsed
   531→    } catch (err) {
   532→      toast({
   533→        title: 'Invalid DOM state JSON',
   534→        description: err instanceof Error ? err.message : String(err),
   535→        variant: 'destructive',
   536→        ...TOAST_DARK,
   537→      })
   538→      return null
   539→    }
   540→  }
   541→
   542→  const refreshState = React.useCallback(async () => {
   543→    try {
   544→      const res = await apexpadApi.getBrowserSessionState(sessionId)
   545→      setState({
   546→        livePageRevision: res.state.livePage?.revision ?? 0,
   547→        refIds: res.state.activeRefMap?.refIds ?? [],
   548→        actionLog: res.state.actionLog,
   549→      })
   550→    } catch {
   551→      /* state probe is best-effort */
   552→    }
   553→  }, [sessionId])
   554→
   555→  React.useEffect(() => {
   556→    void refreshState()
   557→  }, [refreshState])
   558→
   559→  const seed = async () => {
   560→    const elements = parseElements()
   561→    if (!elements) return
   562→    setBusy('seed')
   563→    try {
   564→      const res = await apexpadApi.setLiveDomState({ sessionId, url: url.trim(), elements: elements as never })
   565→      toast({
   566→        title: 'Live DOM state seeded',
   567→        description: `revision ${res.livePage.revision} • ${res.livePage.elementCount} interactive element(s)`,
   568→        ...TOAST_DARK,
   569→      })
   570→      await refreshState()
   571→    } catch (err) {
   572→      toast({
   573→        title: 'Seeding failed',
   574→        description: err instanceof Error ? err.message : String(err),
   575→        variant: 'destructive',
   576→        ...TOAST_DARK,
   577→      })
   578→    } finally {
   579→      setBusy(null)
   580→    }
   581→  }
   582→
   583→  const execute = async () => {
   584→    setBusy('exec')
   585→    const command: AgentBrowserActionCommand =
   586→      commandType === 'fill'
   587→        ? { commandType: 'fill', targetRefOrUrl: targetRef, inputValue }
   588→        : { commandType, targetRefOrUrl: commandType === 'snapshot' || commandType === 'get_url' || commandType === 'close' ? undefined : targetRef }
   589→    try {
   590→      const res = await apexpadApi.executeAgentBrowserCommand({ sessionId, command })
   591→      setLog((prev) => [{ command, result: res.result }, ...prev].slice(0, 12))
   592→      if (!res.result.success) {
   593→        toast({
   594→          title: 'Loud failure (by design)',
   595→          description: res.result.loudFailureReason ?? 'Action blocked by the verification policy.',
   596→          variant: 'destructive',
   597→          ...TOAST_DARK,
   598→        })
   599→      }
   600→      await refreshState()
