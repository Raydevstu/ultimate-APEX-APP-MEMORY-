    66→    66→    const fileSha256 = crypto.createHash('sha256').update(params.content).digest('hex');
    67→    67→    const now = new Date().toISOString();

    69→    69→    // Magic Context Descent: invalidate observations bound to an older blob hash of this file

   199→   199→  /** Ripwire directed call edges + Markdown [[Backlinks]] with Honesty Contract labels. */
   200→   200→  public async rebuildCallGraphEdges(
   201→   201→    projectSlug: string,
   202→   202→    fileContentsByPath: Record<string, string>
   203→   203→  ): Promise<AstDependencyEdge[]> {
   204→   204→    // Merge newly provided contents into the workspace store
   205→   205→    const store = getWorkspaceFileStore();
   206→   206→    for (const [p, c] of Object.entries(fileContentsByPath)) {
   207→   207→      store.set(p, c);
   208→   208→    }
   209→   209→
   210→   210→    const allSymbols = await this.getAllProjectSymbols(projectSlug);
   211→   211→    const symbolsByName = new Map<string, AstSymbolNode[]>();
   212→   212→    for (const sym of allSymbols) {
   213→   213→      const list = symbolsByName.get(sym.symbolName) ?? [];
   214→   214→      list.push(sym);
   215→   215→      symbolsByName.set(sym.symbolName, list);
   216→   216→    }
   217→   217→
   218→   218→    await db.astEdge.deleteMany({ where: { projectSlug } });
   219→   219→
   220→   220→    const edges: AstDependencyEdge[] = [];
   221→   221→    const callTokenRegex = /\b([A-Za-z_][A-Za-z0-9_]{2,})\s*\(/g;
   222→   222→    const wikiLinkRegex = /\[\[([^\]|#]+)(?:#([^\]|]+))?(?:\|[^\]]+)?\]\]/g;
   223→   223→
   224→   224→    for (const callerSym of allSymbols) {
   225→   225→      const fileContent = store.get(callerSym.filePath);
   226→   226→      if (!fileContent) continue;
   227→   227→
   228→   228→      const lines = fileContent.split('\n');
   229→   229→      const bodyText = lines
   230→   230→        .slice(Math.max(0, callerSym.startLine - 1), callerSym.endLine)
   231→   231→        .join('\n');
   232→   232→
   233→   233→      if (callerSym.kind === 'heading') {
   234→   234→        let linkMatch: RegExpExecArray | null;
   235→   235→        while ((linkMatch = wikiLinkRegex.exec(bodyText)) !== null) {
   236→   236→          const targetHeadingOrNote = (linkMatch[2] || linkMatch[1]).trim();
   237→   237→          const candidates = symbolsByName.get(targetHeadingOrNote) ?? [];
   238→   238→          for (const callee of candidates) {
   239→   239→            if (callee.symbolId !== callerSym.symbolId) {
   240→   240→              edges.push(this.makeEdge(projectSlug, callerSym, callee, 'markdown_backlink', 1.0));
   241→   241→            }
   242→   242→          }
   243→   243→        }
   244→   244→      } else {
   245→   245→        const seenCallees = new Set<string>();
   246→   246→        let callMatch: RegExpExecArray | null;
   247→   247→        while ((callMatch = callTokenRegex.exec(bodyText)) !== null) {
   248→   248→          const calledName = callMatch[1];
   249→   249→          if (calledName === callerSym.symbolName) continue;
   250→   250→          const candidates = symbolsByName.get(calledName);
   251→   251→          if (!candidates || candidates.length === 0) continue;
   252→   252→
   253→   253→          // Ripwire Honesty Contract: ambiguous when multiple same-name defs exist
   254→   254→          const honestyLabel: EdgeHonestyLabel =
   255→   255→            candidates.length === 1 ? 'exact' : 'heuristic_ambiguous';
   256→   256→          const splitWeight = Number((1.0 / candidates.length).toFixed(3));
   257→   257→
   258→   258→          for (const callee of candidates) {
   259→   259→            if (callee.symbolId === callerSym.symbolId) continue;
   260→   260→            const dedupeKey = `${callerSym.symbolId}->${callee.symbolId}`;
   261→   261→            if (!seenCallees.has(dedupeKey)) {
   262→   262→              seenCallees.add(dedupeKey);
   263→   263→              edges.push(this.makeEdge(projectSlug, callerSym, callee, honestyLabel, splitWeight));
   264→   264→            }
   265→   265→          }
   266→   266→        }
   267→   267→      }
   268→   268→    }
   269→   269→
   270→   270→    if (edges.length > 0) {
   271→   271→      const callerRows = new Map(allSymbols.map((s) => [s.symbolId, s]));
   272→   272→      const calleeRows = new Map(allSymbols.map((s) => [s.symbolId, s]));
   273→   273→
   274→   274→      await db.astEdge.createMany({
   275→   275→        data: edges.map((e) => ({
   276→   276→          edgeId: e.edgeId,
   277→   277→          projectSlug: e.projectSlug,
   278→   278→          callerSymbolId: e.callerSymbolId,
   279→   279→          calleeSymbolId: e.calleeSymbolId,
   280→   280→          callerFile: callerRows.get(e.callerSymbolId)?.filePath ?? '',
   281→   281→          calleeFile: calleeRows.get(e.calleeSymbolId)?.filePath ?? '',
   282→   282→          honestyLabel: e.honestyLabel,
   283→   283→          weight: e.weight,
   284→   284→        })),
   285→   285→      });
   286→   286→
   287→   287→      // Test-coverage propagation from *.test.*/ *.spec.* callers
   288→   288→      const testCallerEdges = edges.filter((e) => /\.(test|spec)\./i.test(e.callerFile));
   289→   289→      for (const e of testCallerEdges) {
   290→   290→        await db.astSymbol.updateMany({
   291→   291→          where: { symbolId: e.calleeSymbolId, hasTestCoverage: false },
   292→   292→          data: { hasTestCoverage: true },
   293→   293→        });
   294→   294→      }
   295→   295→    }
   296→   296→
   297→   297→    return edges;
   298→   298→  }

   300→   300→  /** Ripwire Personalized PageRank Task-Lens Ranking (alpha=0.85, 20 iterations). */
   301→   301→  public async computePersonalizedPageRank(
   302→   302→    projectSlug: string,
   303→   303→    taskQuery: string,
   304→   304→    iterations = 20,
   305→   305→    damping = 0.85
   306→   306→  ): Promise<AstSymbolNode[]> {
   307→   307→    const symbols = await this.getAllProjectSymbols(projectSlug);
   308→   308→    if (symbols.length === 0) return [];
   309→   309→
   310→   310→    const edges = await this.getAllProjectEdges(projectSlug);
   311→   311→    const n = symbols.length;
   312→   312→    const idToIndex = new Map<string, number>();
   313→   313→    symbols.forEach((s, idx) => idToIndex.set(s.symbolId, idx));
   314→   314→
   315→   315→    const queryTokens = taskQuery
   316→   316→      .toLowerCase()
   317→   317→      .split(/[^a-z0-9_]+/)
   318→   318→      .filter((t) => t.length >= 2);
   319→   319→
   320→   320→    // Seed vector (teleportation distribution)
   321→   321→    const seedVector = new Float64Array(n);
   322→   322→    let seedSum = 0;
   323→   323→
   324→   324→    symbols.forEach((s, idx) => {
   325→   325→      const symLower = s.symbolName.toLowerCase();
   326→   326→      const fileLower = s.filePath.toLowerCase();
   327→   327→      let boost = 0.05; // baseline uniform floor
   328→   328→      for (const tok of queryTokens) {
   329→   329→        if (symLower === tok) boost += 5.0;
   330→   330→        else if (symLower.includes(tok)) boost += 2.0;
   331→   331→        if (fileLower.includes(tok)) boost += 1.0;
   332→   332→      }
   333→   333→      seedVector[idx] = boost;
   334→   334→      seedSum += boost;
   335→   335→    });
   336→   336→
   337→   337→    if (seedSum > 0) {
   338→   338→      for (let i = 0; i < n; i++) {
   339→   339→        seedVector[i] /= seedSum;
   340→   340→      }
   341→   341→    }
   342→   342→
   343→   343→    // Bidirectional adjacency (callers & callees both propagate task relevance)
   344→   344→    const outgoing = Array.from({ length: n }, () => [] as Array<{ to: number; weight: number }>);
   345→   345→    const outWeightSum = new Float64Array(n);
   346→   346→
   347→   347→    for (const e of edges) {
   348→   348→      const u = idToIndex.get(e.callerSymbolId);
   349→   349→      const v = idToIndex.get(e.calleeSymbolId);
   350→   350→      if (u !== undefined && v !== undefined) {
   351→   351→        outgoing[u].push({ to: v, weight: e.weight });
   352→   352→        outWeightSum[u] += e.weight;
   353→   353→        outgoing[v].push({ to: u, weight: e.weight * 0.7 });
   354→   354→        outWeightSum[v] += e.weight * 0.7;
   355→   355→      }
   356→   356→    }
   357→   357→
   358→   358→    let ranks = new Float64Array(seedVector);
   359→   359→    for (let iter = 0; iter < iterations; iter++) {
   360→   360→      const nextRanks = new Float64Array(n);
   361→   361→      let danglingMass = 0;
   362→   362→
   363→   363→      for (let i = 0; i < n; i++) {
   364→   364→        if (outWeightSum[i] === 0) {
   365→   365→          danglingMass += ranks[i];
   366→   366→        } else {
   367→   367→          for (const edge of outgoing[i]) {
   368→   368→            nextRanks[edge.to] += damping * ranks[i] * (edge.weight / outWeightSum[i]);
   369→   369→          }
   370→   370→        }
   371→   371→      }
   372→   372→
   373→   373→      for (let i = 0; i < n; i++) {
   374→   374→        nextRanks[i] +=
   375→   375→          damping * danglingMass * seedVector[i] + (1 - damping) * seedVector[i];
   376→   376→      }
   377→   377→      ranks = nextRanks;
   378→   378→    }
   379→   379→
   380→   380→    const tx: Array<Promise<unknown>> = [];
   381→   381→    symbols.forEach((s, idx) => {
   382→   382→      s.pageRankScore = Number((ranks[idx] * 100).toFixed(4));
   383→   383→      tx.push(
   384→   384→        db.astSymbol.update({
   385→   385→          where: { symbolId: s.symbolId },
   386→   386→          data: { pageRankScore: s.pageRankScore },
   387→   387→        })
   388→   388→      );
   389→   389→    });
   390→   390→    await Promise.all(tx);
   391→   391→
   392→   392→    return [...symbols].sort((a, b) => b.pageRankScore - a.pageRankScore);
   393→   393→  }

   395→   395→  /** Ripwire Blast Radius, Contract Check, Layer Gate & Test Gate. */
   396→   396→  public async analyzeEditBlastRadius(params: {
   397→   397→    projectSlug: string;
   398→   398→    targetFilePath: string;
   399→   399→    proposedContent: string;
   400→   400→    previousContent: string;
   401→   401→  }): Promise<BlastRadiusReport> {
   402→   402→    const allSymbols = await this.getAllProjectSymbols(params.projectSlug);
   403→   403→    const allEdges = await this.getAllProjectEdges(params.projectSlug);
   404→   404→    const symbolMap = new Map<string, AstSymbolNode>(allSymbols.map((s) => [s.symbolId, s]));
   405→   405→
   406→   406→    const fileSymbols = allSymbols.filter((s) => s.filePath === params.targetFilePath);
   407→   407→    const targetSymbolIds = new Set(fileSymbols.map((s) => s.symbolId));
   408→   408→
   409→   409→    // Direct callers
   410→   410→    const directCallerIds = new Set<string>();
   411→   411→    let honestyFloorDisclosed = false;
   412→   412→
   413→   413→    for (const e of allEdges) {
   414→   414→      if (targetSymbolIds.has(e.calleeSymbolId) && !targetSymbolIds.has(e.callerSymbolId)) {
   415→   415→        directCallerIds.add(e.callerSymbolId);
   416→   416→        if (e.honestyLabel === 'heuristic_ambiguous') {
   417→   417→          honestyFloorDisclosed = true;
   418→   418→        }
   419→   419→      }
   420→   420→    }
   421→   421→
   422→   422→    // 2-hop transitive callers
   423→   423→    const transitiveCallerIds = new Set<string>(directCallerIds);
   424→   424→    for (const e of allEdges) {
   425→   425→      if (directCallerIds.has(e.calleeSymbolId) && !targetSymbolIds.has(e.callerSymbolId)) {
   426→   426→        transitiveCallerIds.add(e.callerSymbolId);
   427→   427→      }
   428→   428→    }
   429→   429→
   430→   430→    const directCallers = Array.from(directCallerIds)
   431→   431→      .map((id) => symbolMap.get(id))
   432→   432→      .filter((s): s is AstSymbolNode => Boolean(s));
   433→   433→
   434→   434→    const transitiveCallers = Array.from(transitiveCallerIds)
   435→   435→      .map((id) => symbolMap.get(id))
   436→   436→      .filter((s): s is AstSymbolNode => Boolean(s));
   437→   437→
   438→   438→    const affectedFiles = Array.from(new Set(transitiveCallers.map((s) => s.filePath)));
   439→   439→    const requiredTests = affectedFiles.filter((f) => /\.(test|spec)\./i.test(f));
   440→   440→
   441→   441→    // Contract change detection (old signatures vs proposed content)
   442→   442→    const contractChangesDetected: string[] = [];
   443→   443→    for (const sym of fileSymbols) {
   444→   444→      if (sym.kind !== 'heading' && !params.proposedContent.includes(sym.signature)) {
   445→   445→        contractChangesDetected.push(
   446→   446→          `Signature altered or removed: ${sym.symbolName} (was: "${sym.signature}")`
   447→   447→        );
   448→   448→      }
   449→   449→    }
   450→   450→
   451→   451→    // Architecture layer deny rules from Project DNA (e.g., "deny renderer -> db")
   452→   452→    const dnaRules = await this.memoryRouter.getProjectDnaRules(params.projectSlug);
   453→   453→    const architectureViolations: string[] = [];
   454→   454→    for (const rule of dnaRules) {
   455→   455→      if (rule.layerDenyRule) {
   456→   456→        const match = /^deny\s+([a-zA-Z0-9_-]+)\s*->\s*([a-zA-Z0-9_-]+)$/i.exec(
   457→   457→          rule.layerDenyRule.trim()
   458→   458→        );
   459→   459→        if (match) {
   460→   460→          const fromLayer = match[1].toLowerCase();
   461→   461→          const toLayer = match[2].toLowerCase();

   463→   463→            params.targetFilePath.toLowerCase().includes(fromLayer) &&
   464→   464→            new RegExp(`from\\s+['"][^'"]*${toLayer}[^'"]*['"]`, 'i').test(params.proposedContent)

   468→   468→            );
   469→   469→          }
   470→   470→        }
   471→   471→      }
   472→   472→    }
   473→   473→
   474→   474→    const oldComplexity = this.computeCyclomaticComplexity(params.previousContent);
   475→   475→    const newComplexity = this.computeCyclomaticComplexity(params.proposedContent);
   476→   476→    const cyclomaticDelta = newComplexity - oldComplexity;
   477→   477→    const propagationCost = Number(
   478→   478→      (transitiveCallers.length / Math.max(1, allSymbols.length)).toFixed(4)
   479→   479→    );
   480→   480→
   481→   481→    return {
   482→   482→      targetSymbols: fileSymbols.map((s) => s.symbolName),
   483→   483→      directCallers,
   484→   484→      transitiveCallers,
   485→   485→      affectedFiles,
   486→   486→      requiredTests,
   487→   487→      contractChangesDetected,
   488→   488→      architectureViolations,
   489→   489→      cyclomaticDelta,
   490→   490→      propagationCost,
   491→   491→      honestyFloorDisclosed,
   492→   492→    };
   493→   493→  }

   496→   496→  public async createUiSnapshot(params: {
   497→   497→    sessionId: string;
   498→   498→    url: string;
   499→   499→    elements: Array<{
   500→   500→      role: string;

   518→   518→        role: el.role,

   531→   531→        refObj.isDisabled ? '[disabled]' : '',

   565→   565→        sessionId: params.sessionId,
   566→   566→        url: params.url,
   567→   567→        domHash,
   568→   568→        compactTextTree,
   569→   569→        refMapJson: JSON.stringify(refMap),
   570→   570→        diffFromPrevious,
   571→   571→        createdAt: now,
   572→   572→      },
   573→   573→    });
   574→   574→
   575→   575→    return {
   576→   576→      snapshotId,
   577→   577→      sessionId: params.sessionId,
   578→   578→      url: params.url,
   579→   579→      domHash,
   580→   580→      compactTextTree,
   581→   581→      refMap,
   582→   582→      diffFromPrevious,
   583→   583→      createdAt: now,
   584→   584→    };
   585→   585→  }

   587→   587→  /** Master Context Packer: CacheAligner Frozen Prefix + Byte Ceiling + U-Curve packing. */
   588→   588→  public async buildContextPack(params: {
   589→   589→    sessionId: string;
   590→   590→    projectSlug: string;
   591→   591→    userTaskPrompt: string;
   592→   592→    tokenBudgetCeiling: number;
   593→   593→    activeCcrBlocks?: Array<{ sourceLabel: string; rawContent: string }>;
   594→   594→  }): Promise<PackedContextEnvelope> {

   605→   605→      `  </system_contract>`,

   621→   621→    const byteCeiling = Math.floor(remainingTokenBudget * 2.36 * 0.9);

   626→   626→      params.userTaskPrompt

   632→   632→      query: params.userTaskPrompt,

   636→   636→      indexHits.slice(0, 3).map((h) => h.id)

   671→   671→        topEdgeItems.push(symXml);

   682→   682→      );

   688→   688→      const itemBytes = Buffer.byteLength(obsXml, 'utf8') + 1;
   689→   689→      if (usedBytes + itemBytes <= byteCeiling * 0.65) {
   690→   690→        topEdgeItems.push(obsXml);

   701→   701→        middleTroughItems.push(hitXml);
   702→   702→        includedObservationIds.push(hit.id);
   703→   703→        usedBytes += itemBytes;
   704→   704→      }
   705→   705→    }

   707→   707→    for (const stub of ccrStubs) {
   708→   708→      const stubXml = `    <ccr_payload>\n${stub}\n    </ccr_payload>`;
   709→   709→      const itemBytes = Buffer.byteLength(stubXml, 'utf8') + 1;
   710→   710→      if (usedBytes + itemBytes <= byteCeiling * 0.98) {
   711→   711→        middleTroughItems.push(stubXml);
   712→   712→        usedBytes += itemBytes;
   713→   713→      } else {
   714→   714→        truncationDisclosures.push(`ccr_payload: omitted 1 payload exceeding live-zone byte ceiling`);
   715→   715→      }
   716→   716→    }
   717→   717→
   718→   718→    const liveZoneXml = [
   719→   719→      `<apexpad_live_context byte_ceiling="${byteCeiling}" used_bytes="${usedBytes}">`,
   720→   720→      `  <u_curve_top_high_attention>`,
   721→   721→      ...topEdgeItems,
   722→   722→      `  </u_curve_top_high_attention>`,
   723→   723→      `  <u_curve_middle_progressive_index_and_ccr>`,
   724→   724→      ...middleTroughItems,
   725→   725→      `  </u_curve_middle_progressive_index_and_ccr>`,
   726→   726→      `  <u_curve_bottom_immediate_task>`,

   729→   729→      ...truncationDisclosures.map((d) => `      <disclosure>${d}</disclosure>`),
   730→   730→      `    </honesty_disclosures>`,
   731→   731→      `  </u_curve_bottom_immediate_task>`,
   732→   732→      `</apexpad_live_context>`,
   733→   733→    ].join('\n');
   734→   734→
   735→   735→    const estimatedTotalTokens = frozenTokens + MemoryRouter.estimateTokens(liveZoneXml);
   736→   736→
   737→   737→    return {
   738→   738→      frozenPrefix,
   739→   739→      frozenPrefixHash,
   740→   740→      liveZoneXml,
   741→   741→      estimatedTotalTokens,
   742→   742→      tokenBudgetCeiling: params.tokenBudgetCeiling,
   743→   743→      byteCeiling,
   744→   744→      includedObservationIds,
   745→   745→      includedSymbolIds,
   746→   746→      includedCcrIds,
   747→   747→      truncationDisclosures,
   748→   748→    };

   812→   812→  private findBlockEndLine(lines: string[], startIndex: number): number {
   813→   813→    let depth = 0;
   814→   814→    let foundOpen = false;
   815→   815→    for (let i = startIndex; i < lines.length; i++) {
   816→   816→      const open = (lines[i].match(/\{/g) || []).length;
   817→   817→      const close = (lines[i].match(/\}/g) || []).length;
   818→   818→      if (open > 0) foundOpen = true;
   819→   819→      depth += open - close;
   820→   820→      if (foundOpen && depth <= 0) {
   821→   821→        return i + 1;
   822→   822→      }
   823→   823→    }
   824→   824→    return Math.min(lines.length, startIndex + 25);
   825→   825→  }
